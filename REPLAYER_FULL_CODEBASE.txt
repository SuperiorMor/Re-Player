import com.replayer.ui.screens.HomeScreen
import com.replayer.ui.screens.PlayerScreen
import com.replayer.ui.screens.SettingsScreen
import com.replayer.ui.screens.UpnpBrowseScreen
import com.replayer.ui.theme.AccentCyan
import com.replayer.ui.theme.DarkBackground

class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            RePlayerTheme {
                var selectedNav by remember { mutableStateOf(NavItem.HOME) }
                var activeMediaItem by remember { mutableStateOf<MediaItem?>(null) }
                var showExitDialog by remember { mutableStateOf(false) }
                var initialUpnpFolder by remember { mutableStateOf<Pair<String, String>?>(null) }
                var initialGdriveFolder by remember { mutableStateOf<Pair<String, String>?>(null) }
                var lastPlayedMediaId by remember { mutableStateOf<String?>(null) }

                // Exit confirmation dialog on Home screen
                if (activeMediaItem == null && selectedNav == NavItem.HOME && !showExitDialog) {
                    BackHandler(enabled = true) {
                        showExitDialog = true
                    }
                }

                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(DarkBackground)
                ) {
                    // TV Navigation & Main UI (always kept alive so state is preserved on player close)
                    Row(modifier = Modifier.fillMaxSize()) {
                        TvNavigationRail(
                            selectedItem = selectedNav,
                            onItemSelected = {
                                selectedNav = it
                                initialUpnpFolder = null
                                initialGdriveFolder = null
                            }
                        )

                        Box(modifier = Modifier.weight(1f).fillMaxSize()) {
                            when (selectedNav) {
                                NavItem.HOME -> HomeScreen(
                                    onOpenFavorite = { fav ->
                                        if (fav.sourceType == MediaSourceType.UPNP_DLNA) {
                                            initialUpnpFolder = Pair(fav.id, fav.title)
                                            selectedNav = NavItem.UPNP
                                        } else if (fav.sourceType == MediaSourceType.GOOGLE_DRIVE) {
                                            initialGdriveFolder = Pair(fav.id, fav.title)
                                            selectedNav = NavItem.GDRIVE
                                        }
                                    }
                                )
                                NavItem.UPNP -> UpnpBrowseScreen(
                                    onPlayMedia = {
                                        lastPlayedMediaId = it.id
                                        activeMediaItem = it
                                    },
                                    onNavigateToSettings = { selectedNav = NavItem.SETTINGS },
                                    onBackToHome = { selectedNav = NavItem.HOME },
                                    initialFolder = initialUpnpFolder,
                                    lastPlayedMediaId = lastPlayedMediaId,
                                    isPlayerActive = (activeMediaItem != null),
                                    onFolderLoaded = { initialUpnpFolder = null },
                                    onConsumeLastPlayedMediaId = { lastPlayedMediaId = null }
                                )
                                NavItem.GDRIVE -> GoogleDriveScreen(
                                    onPlayMedia = {
                                        lastPlayedMediaId = it.id
                                        activeMediaItem = it
                                    },
                                    onNavigateToSettings = { selectedNav = NavItem.SETTINGS },
                                    onBackToHome = { selectedNav = NavItem.HOME },
                                    initialFolder = initialGdriveFolder,
                                    lastPlayedMediaId = lastPlayedMediaId,
                                    isPlayerActive = (activeMediaItem != null),
                                    onFolderLoaded = { initialGdriveFolder = null },
                                    onConsumeLastPlayedMediaId = { lastPlayedMediaId = null }
                                )
                                NavItem.SETTINGS -> SettingsScreen(
                                    onBack = { selectedNav = NavItem.HOME }
                                )
                            }
                        }
                    }

                    // Fullscreen Player Overlay
                    if (activeMediaItem != null) {
                        PlayerScreen(
                            mediaItem = activeMediaItem!!,
                            onBack = { activeMediaItem = null }
                        )
                    }

                    // TV Exit Confirmation Dialog
                    if (showExitDialog) {
                        BackHandler(enabled = true) {
                            showExitDialog = false
                        }

                        Dialog(onDismissRequest = { showExitDialog = false }) {
                            Box(
                                modifier = Modifier
                                    .width(440.dp)
                                    .clip(RoundedCornerShape(20.dp))
                                    .background(DarkCard)
                                    .border(1.dp, AccentCyan.copy(alpha = 0.5f), RoundedCornerShape(20.dp))
                                    .padding(28.dp)
                            ) {
                                Column(
                                    horizontalAlignment = Alignment.CenterHorizontally,
                                    verticalArrangement = Arrangement.Center
                                ) {
                                    Icon(
                                        Icons.Default.ExitToApp,
                                        contentDescription = null,
                                        tint = AccentCyan,
                                        modifier = Modifier.size(44.dp)
                                    )
                                    Spacer(modifier = Modifier.height(16.dp))
                                    Text(
                                        text = "Выход из Re:Player",
                                        style = MaterialTheme.typography.titleLarge.copy(fontWeight = FontWeight.Bold),
                                        color = TextPrimary
                                    )
                                    Spacer(modifier = Modifier.height(8.dp))
                                    Text(
                                        text = "Вы точно хотите закрыть приложение?",
                                        style = MaterialTheme.typography.bodyLarge,
                                        color = TextSecondary,
                                        textAlign = TextAlign.Center
                                    )
                                    Spacer(modifier = Modifier.height(24.dp))
                                    Row(
                                        modifier = Modifier.fillMaxWidth(),
                                        horizontalArrangement = Arrangement.spacedBy(16.dp)
                                    ) {
                                        TvFocusCard(
                                            onClick = {
                                                showExitDialog = false
                                                this@MainActivity.finish()
                                            },
                                            modifier = Modifier
                                                .weight(1f)
                                                .height(52.dp)
                                        ) {
                                            Box(
                                                modifier = Modifier.fillMaxSize(),
                                                contentAlignment = Alignment.Center
                                            ) {
                                                Text(
                                                    text = "Да, выйти",
                                                    style = MaterialTheme.typography.titleMedium,
                                                    color = Color(0xFFFF5252)
                                                )
                                            }
                                        }

                                        TvFocusCard(
                                            onClick = { showExitDialog = false },
                                            modifier = Modifier
                                                .weight(1f)
                                                .height(52.dp)
                                        ) {
                                            Box(
                                                modifier = Modifier.fillMaxSize(),
                                                contentAlignment = Alignment.Center
                                            ) {
                                                Text(
                                                    text = "Отмена",
                                                    style = MaterialTheme.typography.titleMedium,
                                                    color = TextPrimary
                                                )
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}

package com.replayer.model

import kotlinx.serialization.Serializable
import java.util.Locale

@Serializable
data class MediaItem(
    val id: String,
    val title: String,
    val uri: String,
    val type: MediaType,
    val sourceType: MediaSourceType,
    val size: Long = 0L,
    val durationMs: Long = 0L,
    val mimeType: String? = null,
    val thumbnailUrl: String? = null,
    val parentId: String? = null,
    val dateAdded: String? = null,
    val format: String? = null
) {
    fun formattedSize(): String {
        if (type == MediaType.DIRECTORY || size <= 0L) return ""
        val kb = size / 1024.0
        val mb = kb / 1024.0
        val gb = mb / 1024.0
        return when {
            gb >= 1.0 -> String.format(Locale.US, "%.2f ГБ", gb)
            mb >= 1.0 -> String.format(Locale.US, "%.1f МБ", mb)
            kb >= 1.0 -> String.format(Locale.US, "%.0f КБ", kb)
            else -> "$size Б"
        }
    }

    fun fileFormat(): String {
        if (type == MediaType.DIRECTORY) return "ПАПКА"
        if (!format.isNullOrEmpty()) return format.uppercase()
        val ext = title.substringAfterLast('.', "")
        return if (ext.isNotEmpty() && ext.length <= 5) ext.uppercase() else type.name
    }

    fun formattedDate(): String {
        if (dateAdded.isNullOrEmpty()) return ""
        // If it's ISO like 2023-05-12T14:30:00 -> take 2023-05-12
        return if (dateAdded.contains('T')) {
            dateAdded.substringBefore('T')
        } else {
            dateAdded.take(10)
        }
    }
}

class ServerConfigManager(context: Context) {

    private val prefs: SharedPreferences =
        context.getSharedPreferences("replayer_server_config", Context.MODE_PRIVATE)

    private val json = Json { ignoreUnknownKeys = true }

    private val _connectedUpnpDevice = MutableStateFlow<UpnpDevice?>(loadConnectedDevice())
    val connectedUpnpDevice: StateFlow<UpnpDevice?> = _connectedUpnpDevice.asStateFlow()

    private fun loadConnectedDevice(): UpnpDevice? {
        val raw = prefs.getString("connected_upnp_device", null) ?: return null
        return try {
            json.decodeFromString<UpnpDevice>(raw)
        } catch (_: Exception) {
            null
        }
    }

        _connectedUpnpDevice.value = device
        try {
            val encoded = json.encodeToString(device)
            prefs.edit().putString("connected_upnp_device", encoded).apply()
        } catch (_: Exception) {}
    }

    fun disconnectUpnpDevice() {
        _connectedUpnpDevice.value = null
        prefs.edit().remove("connected_upnp_device").apply()
    }
}

package com.replayer.data

import android.content.Context
import android.content.SharedPreferences
import com.replayer.model.FavoriteFolder
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json

class FavoritesManager(context: Context) {

    private val prefs: SharedPreferences =
        context.getSharedPreferences("replayer_favorites", Context.MODE_PRIVATE)

    private val json = Json { ignoreUnknownKeys = true }

    private val _favorites = MutableStateFlow<List<FavoriteFolder>>(loadFavorites())
    val favorites: StateFlow<List<FavoriteFolder>> = _favorites.asStateFlow()

    private fun loadFavorites(): List<FavoriteFolder> {
        val raw = prefs.getString("favorites_list", null) ?: return emptyList()
        return try {
            json.decodeFromString<List<FavoriteFolder>>(raw)
        } catch (_: Exception) {
            emptyList()
        }
    }

    private fun saveFavorites(list: List<FavoriteFolder>) {
        _favorites.value = list
        try {
            val encoded = json.encodeToString(list)
            prefs.edit().putString("favorites_list", encoded).apply()
        } catch (_: Exception) {}
    }

    fun isFavorite(id: String): Boolean {
        return _favorites.value.any { it.id == id }
    }

    fun toggleFavorite(folder: FavoriteFolder) {
        val current = _favorites.value.toMutableList()
        val index = current.indexOfFirst { it.id == folder.id }
        if (index >= 0) {
            current.removeAt(index)
        } else {
            current.add(folder)
        }
        saveFavorites(current)
    }

    fun removeFavorite(id: String) {
        val current = _favorites.value.filterNot { it.id == id }
        saveFavorites(current)
    }
}

package com.replayer.network.upnp

import android.content.Context
import android.net.wifi.WifiManager
import android.util.Log
import com.replayer.model.UpnpDevice
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.flow.flowOn
import kotlinx.coroutines.withContext
import java.net.DatagramPacket
import java.net.DatagramSocket
import java.net.InetAddress
import java.net.SocketTimeoutException

class SsdpDiscovery(private val context: Context) {

    companion object {
        private const val TAG = "SsdpDiscovery"
        private const val SSDP_IP = "239.255.255.250"
        private const val SSDP_PORT = 1900
        private const val SEARCH_TARGET = "urn:schemas-upnp-org:service:ContentDirectory:1"
    }

    fun discoverDevices(timeoutMs: Int = 4000): Flow<List<UpnpDevice>> = flow {
        val wifiManager = context.applicationContext.getSystemService(Context.WIFI_SERVICE) as WifiManager
        val multicastLock = wifiManager.createMulticastLock("RePlayerSsdpLock").apply {
            setReferenceCounted(true)
            acquire()
        }

        val discoveredMap = mutableMapOf<String, UpnpDevice>()

        try {
            val searchMessage = buildString {
                append("M-SEARCH * HTTP/1.1\r\n")
                append("HOST: $SSDP_IP:$SSDP_PORT\r\n")
                append("MAN: \"ssdp:discover\"\r\n")
                append("MX: 3\r\n")
                append("ST: $SEARCH_TARGET\r\n")
                append("\r\n")
            }

            val socket = DatagramSocket()
            socket.soTimeout = timeoutMs

            val sendData = searchMessage.toByteArray()
            val group = InetAddress.getByName(SSDP_IP)
            val packet = DatagramPacket(sendData, sendData.size, group, SSDP_PORT)
            socket.send(packet)

            val receiveBuffer = ByteArray(4096)
            val startTime = System.currentTimeMillis()

            while (System.currentTimeMillis() - startTime < timeoutMs) {
                try {
                    val receivePacket = DatagramPacket(receiveBuffer, receiveBuffer.size)
                    socket.receive(receivePacket)
                    val response = String(receivePacket.data, 0, receivePacket.length)

                    val location = extractHeader(response, "LOCATION")
                    val usn = extractHeader(response, "USN") ?: location

                    if (location != null && !discoveredMap.containsKey(location)) {
                        val device = UpnpDeviceParser.fetchAndParse(location)
                        if (device != null) {
                            discoveredMap[location] = device
                            emit(discoveredMap.values.toList())
                        }
                    }
                } catch (e: SocketTimeoutException) {
                    break
                } catch (e: Exception) {
                    Log.w(TAG, "Error receiving SSDP packet", e)
                }
            }
            socket.close()
        } finally {
            if (multicastLock.isHeld) {
                multicastLock.release()
            }
        }
        emit(discoveredMap.values.toList())
    }.flowOn(Dispatchers.IO)

    private fun extractHeader(response: String, headerName: String): String? {
        val lines = response.split("\r?\n".toRegex())
        for (line in lines) {
            val parts = line.split(":", limit = 2)
            if (parts.size == 2 && parts[0].trim().equals(headerName, ignoreCase = true)) {
                return parts[1].trim()
            }
        }
        return null
    }
}

package com.replayer.network.upnp

import android.util.Log
import android.util.Xml
import com.replayer.model.UpnpDevice
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.OkHttpClient
import okhttp3.Request
import org.xmlpull.v1.XmlPullParser
import java.io.StringReader
import java.net.URL
import java.util.concurrent.TimeUnit

object UpnpDeviceParser {
    private const val TAG = "UpnpDeviceParser"
    private val client = OkHttpClient.Builder()
        .connectTimeout(3, TimeUnit.SECONDS)
        .readTimeout(3, TimeUnit.SECONDS)
        .build()

    suspend fun fetchAndParse(locationUrl: String): UpnpDevice? = withContext(Dispatchers.IO) {
        try {
            val request = Request.Builder().url(locationUrl).build()
            val response = client.newCall(request).execute()
            if (!response.isSuccessful) return@withContext null
            val body = response.body?.string() ?: return@withContext null

            val parser = Xml.newPullParser()
            parser.setInput(StringReader(body))

            var friendlyName = "DLNA Media Server"
            var manufacturer = "Unknown"
            var udn = locationUrl
            var contentDirectoryControlUrl: String? = null
            var insideContentDirectory = false
            var currentTag = ""

            var eventType = parser.eventType
            while (eventType != XmlPullParser.END_DOCUMENT) {
                when (eventType) {
                    XmlPullParser.START_TAG -> {
                        currentTag = parser.name
                        if (currentTag == "service") {
                            insideContentDirectory = false
                        }
                    }
                    XmlPullParser.TEXT -> {
                        val text = parser.text.trim()
                        if (text.isNotEmpty()) {
                            when (currentTag) {
                                "friendlyName" -> friendlyName = text
                                "manufacturer" -> manufacturer = text
                                "UDN" -> udn = text
                                "serviceType" -> {
                                    if (text.contains("ContentDirectory")) {
                                        insideContentDirectory = true
                                    }
                                }
                                "controlURL" -> {
                                    if (insideContentDirectory && contentDirectoryControlUrl == null) {
                                        contentDirectoryControlUrl = resolveUrl(locationUrl, text)
                                    }
                                }
                            }
                        }
                    }
                    XmlPullParser.END_TAG -> {
                        if (parser.name == "service") {
                            insideContentDirectory = false
                        }
                    }
                }
                eventType = parser.next()
            }

            UpnpDevice(
                id = udn,
                friendlyName = friendlyName,
                manufacturer = manufacturer,
                location = locationUrl,
                contentDirectoryControlUrl = contentDirectoryControlUrl
            )
        } catch (e: Exception) {
            Log.e(TAG, "Error parsing device description for $locationUrl", e)
            null
        }
    }

    fun resolveUrl(baseUrl: String, path: String): String {
        return if (path.startsWith("http://") || path.startsWith("https://")) {
            path
        } else {
            val base = URL(baseUrl)
            URL(base, path).toString()
        }
    }
}

package com.replayer.network.upnp

import android.util.Log
import android.util.Xml
import com.replayer.model.MediaItem
import com.replayer.model.MediaSourceType
import com.replayer.model.MediaType
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.xmlpull.v1.XmlPullParser
import java.io.StringReader

class ContentDirectoryService {
    private val client = OkHttpClient()

    suspend fun browse(controlUrl: String, objectId: String = "0"): List<MediaItem> = withContext(Dispatchers.IO) {
        val soapAction = "\"urn:schemas-upnp-org:service:ContentDirectory:1#Browse\""
        val soapBody = buildString {
            append("<?xml version=\"1.0\" encoding=\"utf-8\"?>")
            append("<s:Envelope xmlns:s=\"http://schemas.xmlsoap.org/soap/envelope/\" s:encodingStyle=\"http://schemas.xmlsoap.org/soap/encoding/\">")
            append("<s:Body>")
            append("<u:Browse xmlns:u=\"urn:schemas-upnp-org:service:ContentDirectory:1\">")
            append("<ObjectID>$objectId</ObjectID>")
            append("<BrowseFlag>BrowseDirectChildren</BrowseFlag>")
            append("<Filter>*</Filter>")
            append("<StartingIndex>0</StartingIndex>")
            append("<RequestedCount>100</RequestedCount>")
            append("<SortCriteria></SortCriteria>")
            append("</u:Browse>")
            append("</s:Body>")
            append("</s:Envelope>")
        }

        val request = Request.Builder()
            .url(controlUrl)
            .addHeader("SOAPAction", soapAction)
            .post(soapBody.toRequestBody("text/xml; charset=utf-8".toMediaType()))
            .build()

        val items = mutableListOf<MediaItem>()

        try {
            val response = client.newCall(request).execute()
            if (!response.isSuccessful) return@withContext emptyList()
            val rawXml = response.body?.string() ?: return@withContext emptyList()

            val resultXml = extractResultTag(rawXml) ?: return@withContext emptyList()
            parseDidlLite(resultXml, items, objectId)
        } catch (e: Exception) {
            Log.e("ContentDirectoryService", "Browse error for $controlUrl", e)
        }

        items
    }

    private fun extractResultTag(soapXml: String): String? {
        val startIdx = soapXml.indexOf("<Result")
        if (startIdx == -1) return null
        val tagCloseIdx = soapXml.indexOf(">", startIdx)
        if (tagCloseIdx == -1) return null
        val endIdx = soapXml.indexOf("</Result>", tagCloseIdx)
        if (endIdx == -1) return null

        val encoded = soapXml.substring(tagCloseIdx + 1, endIdx)
        return encoded.replace("&lt;", "<")
            .replace("&gt;", ">")
            .replace("&amp;", "&")
            .replace("&quot;", "\"")
            .replace("&apos;", "'")
    }

    private fun parseDidlLite(didlXml: String, items: MutableList<MediaItem>, parentId: String) {
        val parser = Xml.newPullParser()
        parser.setInput(StringReader(didlXml))

        var eventType = parser.eventType
        var currentItem: MutableMap<String, String>? = null
        val textBuffer = StringBuilder()

        while (eventType != XmlPullParser.END_DOCUMENT) {
            when (eventType) {
                XmlPullParser.START_TAG -> {
                    val rawName = parser.name ?: ""
                    val localTag = rawName.substringAfterLast(':').lowercase()
                    textBuffer.clear()

                    if (localTag == "container" || localTag == "item") {
                        currentItem = mutableMapOf()
                        currentItem["isContainer"] = if (localTag == "container") "true" else "false"
                        currentItem["id"] = parser.getAttributeValue(null, "id") ?: ""
                    }
                    if (localTag == "res" && currentItem != null) {
                        val sizeAttr = parser.getAttributeValue(null, "size")
                        if (!sizeAttr.isNullOrEmpty()) {
                            currentItem["size"] = sizeAttr
                        }
                    }
                }
                XmlPullParser.TEXT -> {
                    textBuffer.append(parser.text ?: "")
                }
                XmlPullParser.END_TAG -> {
                    val rawName = parser.name ?: ""
                    val localTag = rawName.substringAfterLast(':').lowercase()
                    val text = textBuffer.toString().trim()

                    if (currentItem != null && text.isNotEmpty()) {
                        when (localTag) {
                            "title" -> currentItem["title"] = text
                            "res" -> currentItem["res"] = text
                            "class" -> currentItem["class"] = text
                            "date" -> currentItem["date"] = text
                            "albumarturi" -> currentItem["albumArt"] = text
                        }
                    }

                    if ((localTag == "container" || localTag == "item") && currentItem != null) {
                        val isContainer = currentItem["isContainer"] == "true"
                        val upnpClass = currentItem["class"] ?: ""
                        val title = currentItem["title"]?.trim()?.ifEmpty { "Unnamed" } ?: "Unnamed"
                        val id = currentItem["id"] ?: ""
                        val res = currentItem["res"] ?: ""
                        val size = currentItem["size"]?.toLongOrNull() ?: 0L
                        val date = currentItem["date"]
                        val ext = title.substringAfterLast('.', "").uppercase()
                        val format = if (ext.isNotEmpty() && ext.length <= 5) ext else null

                        val mediaType = when {
                            isContainer -> MediaType.DIRECTORY
                            upnpClass.contains("videoItem", ignoreCase = true) -> MediaType.VIDEO
                            upnpClass.contains("audioItem", ignoreCase = true) -> MediaType.AUDIO
                            upnpClass.contains("imageItem", ignoreCase = true) -> MediaType.IMAGE
                            res.endsWith(".mkv", ignoreCase = true) || res.endsWith(".mp4", ignoreCase = true) ||
                            res.endsWith(".avi", ignoreCase = true) || res.endsWith(".ts", ignoreCase = true) ||
                            res.endsWith(".m2ts", ignoreCase = true) || res.endsWith(".mov", ignoreCase = true) ||
                            res.endsWith(".webm", ignoreCase = true) -> MediaType.VIDEO
                            res.endsWith(".mp3", ignoreCase = true) || res.endsWith(".flac", ignoreCase = true) ||
                            res.endsWith(".m4a", ignoreCase = true) || res.endsWith(".wav", ignoreCase = true) ||
                            res.endsWith(".aac", ignoreCase = true) || res.endsWith(".ogg", ignoreCase = true) -> MediaType.AUDIO
                            res.endsWith(".jpg", ignoreCase = true) || res.endsWith(".jpeg", ignoreCase = true) ||
                            res.endsWith(".png", ignoreCase = true) || res.endsWith(".webp", ignoreCase = true) -> MediaType.IMAGE
                            else -> MediaType.VIDEO
                        }

                        items.add(
                            MediaItem(
                                id = id,
                                title = title,
                                uri = res,
                                type = mediaType,
                                sourceType = MediaSourceType.UPNP_DLNA,
                                size = size,
                                parentId = parentId,
                                dateAdded = date,
                                format = format
                            )
                        )
                        currentItem = null
                    }
                    textBuffer.clear()
                }
            }
            eventType = parser.next()
        }
    }
}

    suspend fun refreshAccessToken(): String? = withContext(Dispatchers.IO) {
        try {
            val email = getSavedAccountEmail()
            if (!email.isNullOrBlank()) {
                val oldToken = getSavedAccessToken()
                if (oldToken != null) {
                    try {
                        GoogleAuthUtil.clearToken(context, oldToken)
                    } catch (_: Exception) {}
                }
                val account = Account(email, "com.google")
                val newToken = GoogleAuthUtil.getToken(
                    context,
                    account,
                    "oauth2:https://www.googleapis.com/auth/drive.readonly"
                )
                if (!newToken.isNullOrBlank()) {
                    saveTokens(newToken, null, expiresInSec = 3600)
                    return@withContext newToken
                }
            }
        } catch (e: Exception) {
            android.util.Log.e("DriveAuth", "Refresh via GoogleAuthUtil failed: ${e.message}")
        }

        val refreshToken = prefs.getString("refresh_token", null) ?: return@withContext null
        val clientId = getSavedClientId()
        val clientSecret = getSavedClientSecret()
        if (clientId.isBlank() || clientSecret.isBlank()) return@withContext null

        val form = FormBody.Builder()
            .add("client_id", clientId.trim())
            .add("client_secret", clientSecret.trim())
            .add("refresh_token", refreshToken)
            .add("grant_type", "refresh_token")
            .build()

        val request = Request.Builder().url(TOKEN_URL).post(form).build()
        try {
            val response = client.newCall(request).execute()
            val body = response.body?.string() ?: return@withContext null
            if (!response.isSuccessful) {
                android.util.Log.e("DriveAuth", "Refresh failed (${response.code}): $body")
                return@withContext null
            }
            val tokenResp = json.decodeFromString<TokenResponse>(body)
            if (!tokenResp.access_token.isNullOrBlank()) {
                saveTokens(tokenResp.access_token, tokenResp.refresh_token ?: refreshToken, tokenResp.expires_in)
                return@withContext tokenResp.access_token
            }
        } catch (e: Exception) {
            android.util.Log.e("DriveAuth", "Refresh exception: ${e.message}")
        }
        null
    }

    suspend fun requestDeviceCode(clientId: String = getSavedClientId()): Result<DeviceCodeResponse> = withContext(Dispatchers.IO) {
        val actualClientId = clientId.ifBlank { getSavedClientId() }
        if (actualClientId.isBlank() || actualClientId.contains("YOUR_TV_CLIENT_ID")) {
            return@withContext Result.failure(
                Exception("Client ID не настроен.")
            )
        }

        val formPrimary = FormBody.Builder()
            .add("client_id", actualClientId.trim())
            .add("scope", PRIMARY_SCOPE)
            .build()

        val requestPrimary = Request.Builder().url(DEVICE_CODE_URL).post(formPrimary).build()
        try {
            val responsePrimary = client.newCall(requestPrimary).execute()
            val bodyPrimary = responsePrimary.body?.string() ?: ""
            if (responsePrimary.isSuccessful) {
                val codeResp = json.decodeFromString<DeviceCodeResponse>(bodyPrimary)
                saveClientId(actualClientId)
                return@withContext Result.success(codeResp)
            }

            // If invalid_scope, gracefully retry with fallback scope
            if (bodyPrimary.contains("invalid_scope")) {
                val formFallback = FormBody.Builder()
                    .add("client_id", actualClientId.trim())
                    .add("scope", FALLBACK_SCOPE)
                    .build()
                val requestFallback = Request.Builder().url(DEVICE_CODE_URL).post(formFallback).build()
                val responseFallback = client.newCall(requestFallback).execute()
                val bodyFallback = responseFallback.body?.string() ?: ""
                if (responseFallback.isSuccessful) {
                    val codeResp = json.decodeFromString<DeviceCodeResponse>(bodyFallback)
                    saveClientId(actualClientId)
                    return@withContext Result.success(codeResp)
                }
                return@withContext Result.failure(Exception("Ошибка Google API (${responseFallback.code}): $bodyFallback"))
            }

            return@withContext Result.failure(Exception("Ошибка Google API (${responsePrimary.code}): $bodyPrimary"))
        } catch (e: Exception) {
            Result.failure(e)
        }
    }

    fun isAuthenticated(): Boolean = !getSavedAccessToken().isNullOrBlank()

    suspend fun pollForToken(
        clientId: String,
        deviceCode: String,
        clientSecret: String = "",
        intervalSec: Int = 5,
        maxRetries: Int = 120
    ): TokenResponse? = withContext(Dispatchers.IO) {
        val effectiveSecret = clientSecret.ifBlank { getSavedClientSecret() }
        val formBuilder = FormBody.Builder()
            .add("client_id", clientId.trim())
            .add("device_code", deviceCode)
            .add("grant_type", "urn:ietf:params:oauth:grant-type:device_code")
        if (effectiveSecret.isNotBlank()) {
            formBuilder.add("client_secret", effectiveSecret.trim())
        }
        val form = formBuilder.build()

        var currentInterval = intervalSec.coerceAtLeast(5)
        var retries = 0
        while (retries < maxRetries) {
            delay(currentInterval * 1000L)
            retries++

            val request = Request.Builder().url(TOKEN_URL).post(form).build()
            try {
                val response = client.newCall(request).execute()
                val body = response.body?.string() ?: continue
                android.util.Log.d("DriveAuth", "Poll ($retries): code=${response.code}, body=$body")

                val tokenResp = try {
                    json.decodeFromString<TokenResponse>(body)
                } catch (e: Exception) {
                    android.util.Log.w("DriveAuth", "Poll JSON parse error: ${e.message}")
                    null
                }

                if (tokenResp != null && !tokenResp.access_token.isNullOrBlank()) {
                    saveTokens(tokenResp.access_token, tokenResp.refresh_token, tokenResp.expires_in)
                    return@withContext tokenResp
                }

                val errorCode = tokenResp?.error
                when (errorCode) {
                    "authorization_pending" -> {
                        // User is still authorizing on phone / browser; keep polling
                    }
                    "slow_down" -> {
                        // Rate limit: increase polling interval by 5 seconds
                        currentInterval += 5
                    }
                    "access_denied", "expired_token" -> {
                        return@withContext tokenResp
                    }
                    else -> {
                        if (errorCode != null) {
                            return@withContext tokenResp
                        }
                    }
                }
            } catch (e: Exception) {
                android.util.Log.w("DriveAuth", "Poll exception: ${e.message}")
            }
        }
        null
    }
}

    suspend fun getOrCreateRePlayerFolder(token: String): MediaItem? = withContext(Dispatchers.IO) {
        val searchUrl = "https://www.googleapis.com/drive/v3/files".toHttpUrl().newBuilder()
            .addQueryParameter("q", "mimeType = 'application/vnd.google-apps.folder' and name = 'Re:Player' and trashed = false")
            .addQueryParameter("fields", "files(id,name,mimeType)")
            .addQueryParameter("pageSize", "1")
            .addQueryParameter("supportsAllDrives", "true")
            .addQueryParameter("includeItemsFromAllDrives", "true")
            .build()

        val searchReq = Request.Builder()
            .url(searchUrl)
            .addHeader("Authorization", "Bearer $token")
            .build()

        try {
            val response = client.newCall(searchReq).execute()
            val body = response.body?.string() ?: ""
            if (response.isSuccessful) {
                val rootObj = json.parseToJsonElement(body).jsonObject
                val files = rootObj["files"]?.jsonArray
                if (files != null && files.isNotEmpty()) {
                    val folderObj = files[0].jsonObject
                    val id = folderObj["id"]?.jsonPrimitive?.content ?: ""
                    if (id.isNotBlank()) {
                        return@withContext MediaItem(
                            id = id,
                            title = "Re:Player",
                            uri = "",
                            type = MediaType.DIRECTORY,
                            sourceType = MediaSourceType.GOOGLE_DRIVE,
                            parentId = "root"
                        )
                    }
                }
            }

            // Folder does not exist, create it
            val createUrl = "https://www.googleapis.com/drive/v3/files".toHttpUrl().newBuilder()
                .addQueryParameter("supportsAllDrives", "true")
                .build()

            val jsonBody = """{"name":"Re:Player","mimeType":"application/vnd.google-apps.folder"}"""
            val reqBody = jsonBody.toRequestBody("application/json; charset=utf-8".toMediaType())

            val createReq = Request.Builder()
                .url(createUrl)
                .addHeader("Authorization", "Bearer $token")
                .post(reqBody)
                .build()

            val createResp = client.newCall(createReq).execute()
            val createBody = createResp.body?.string() ?: ""
            if (createResp.isSuccessful) {
                val createdObj = json.parseToJsonElement(createBody).jsonObject
                val id = createdObj["id"]?.jsonPrimitive?.content ?: ""
                if (id.isNotBlank()) {
                    return@withContext MediaItem(
                        id = id,
                        title = "Re:Player",
                        uri = "",
                        type = MediaType.DIRECTORY,
                        sourceType = MediaSourceType.GOOGLE_DRIVE,
                        parentId = "root"
                    )
                }
            } else {
                Log.e("DriveService", "Create folder error (${createResp.code}): $createBody")
            }
        } catch (e: Exception) {
            Log.e("DriveService", "Error getOrCreateRePlayerFolder: ${e.message}", e)
        }
        null
    }

    suspend fun listFolder(folderId: String = "root"): List<MediaItem> = withContext(Dispatchers.IO) {
        lastErrorMessage = null
        var token = authManager.getValidAccessToken() ?: authManager.getSavedAccessToken()
        if (token.isNullOrBlank()) {
            lastErrorMessage = "Требуется авторизация"
            return@withContext emptyList()
        }

        val isRoot = folderId == "root"
        val primaryQuery = if (isRoot) {
            "'root' in parents and trashed = false"
        } else {
            "'$folderId' in parents and trashed = false"
        }

        var items = executeDriveQuery(token, primaryQuery, folderId).toMutableList()

        // If token was expired (401), try refreshing token and re-execute
        if (lastErrorMessage?.contains("401") == true) {
            val newToken = authManager.refreshAccessToken()
            if (!newToken.isNullOrBlank()) {
                token = newToken
                lastErrorMessage = null
                items = executeDriveQuery(token, primaryQuery, folderId).toMutableList()
            }
        }

        // If root has no items with "'root' in parents", fallback to broad query for accessible files/folders
        if (isRoot && items.isEmpty() && lastErrorMessage == null) {
            val broadItems = executeDriveQuery(token, "trashed = false", folderId)
            val existingIds = items.map { it.id }.toSet()
            for (item in broadItems) {
                if (item.id !in existingIds) {
                    items.add(item)
                }
            }
        }

        items
    }

    private fun executeDriveQuery(token: String, query: String, parentFolderId: String): List<MediaItem> {
        val httpUrl = "https://www.googleapis.com/drive/v3/files".toHttpUrl().newBuilder()
            .addQueryParameter("q", query)
            .addQueryParameter("fields", "files(id,name,mimeType,size,thumbnailLink,parents)")
            .addQueryParameter("pageSize", "100")
            .addQueryParameter("supportsAllDrives", "true")
            .addQueryParameter("includeItemsFromAllDrives", "true")
            .build()

        val request = Request.Builder()
            .url(httpUrl)
            .addHeader("Authorization", "Bearer $token")
            .build()

        val items = mutableListOf<MediaItem>()
        try {
            val response = client.newCall(request).execute()
            val body = response.body?.string() ?: ""
            if (!response.isSuccessful) {
                Log.e("DriveService", "Drive API error (${response.code}): $body")
                lastErrorMessage = "Ошибка Google Drive (${response.code})"
                return emptyList()
            }

            val rootObj = json.parseToJsonElement(body).jsonObject
            val filesArray = rootObj["files"]?.jsonArray ?: return emptyList()

            for (element in filesArray) {
                val f = element.jsonObject
                val id = f["id"]?.jsonPrimitive?.content ?: continue
                val name = f["name"]?.jsonPrimitive?.content ?: "Unnamed"
                val mimeType = f["mimeType"]?.jsonPrimitive?.content ?: ""
                val size = f["size"]?.jsonPrimitive?.content?.toLongOrNull() ?: 0L
                val thumb = f["thumbnailLink"]?.jsonPrimitive?.content

                val isFolder = mimeType == "application/vnd.google-apps.folder"
                val lowerName = name.lowercase()
                val mediaType = when {
                    isFolder -> MediaType.DIRECTORY
                    mimeType.startsWith("video/") -> MediaType.VIDEO
                    mimeType.startsWith("audio/") -> MediaType.AUDIO
                    mimeType.startsWith("image/") -> MediaType.IMAGE
                    lowerName.endsWith(".mkv") || lowerName.endsWith(".avi") || lowerName.endsWith(".mp4") ||
                    lowerName.endsWith(".mov") || lowerName.endsWith(".ts") || lowerName.endsWith(".m2ts") ||
                    lowerName.endsWith(".webm") || lowerName.endsWith(".flv") || lowerName.endsWith(".wmv") -> MediaType.VIDEO
                    lowerName.endsWith(".mp3") || lowerName.endsWith(".flac") || lowerName.endsWith(".wav") ||
                    lowerName.endsWith(".m4a") || lowerName.endsWith(".aac") || lowerName.endsWith(".ogg") -> MediaType.AUDIO
                    lowerName.endsWith(".jpg") || lowerName.endsWith(".jpeg") || lowerName.endsWith(".png") ||
                    lowerName.endsWith(".webp") -> MediaType.IMAGE
                    else -> continue
                }

                val ext = name.substringAfterLast('.', "").uppercase()
                val format = if (ext.isNotEmpty() && ext.length <= 5) ext else null

                // Streaming URL with alt=media and acknowledgeAbuse for large video files
                val streamUrl = "https://www.googleapis.com/drive/v3/files/$id?alt=media&acknowledgeAbuse=true"

                items.add(
                    MediaItem(
                        id = id,
                        title = name,
                        uri = streamUrl,
                        type = mediaType,
                        sourceType = MediaSourceType.GOOGLE_DRIVE,
                        size = size,
                        mimeType = mimeType,
                        thumbnailUrl = thumb,
                        parentId = parentFolderId,
                        format = format
                    )
                )
            }
        } catch (e: Exception) {
            Log.e("DriveService", "Exception in executeDriveQuery: ${e.message}", e)
            lastErrorMessage = e.localizedMessage ?: "Ошибка сети"
        }
        return items
    }
}

import java.net.Socket
import java.net.URLDecoder
import java.util.UUID

class LocalPairingServer(
    private val context: Context,
    private val authManager: GoogleDriveAuthManager,
    private val onPaired: (String?) -> Unit
) {
    private val scope = CoroutineScope(Dispatchers.IO)
    private var serverSocket: ServerSocket? = null
    private var isRunning = false
    private var pollJob: Job? = null
    private val okClient = OkHttpClient()
    private val json = Json { ignoreUnknownKeys = true }

    val sessionId: String = UUID.randomUUID().toString().take(8).lowercase()
    var localIp: String = "127.0.0.1"
        private set
    var port: Int = 8765
        private set

    val pairingUrl: String
        get() = "http://$localIp:$port/?session=$sessionId"

    fun start() {
        if (isRunning) return
        isRunning = true
        localIp = getLocalIpAddress()

        // 1. Start local TCP HTTP Server
        scope.launch {
            try {
                serverSocket = ServerSocket(port)
            } catch (e: Exception) {
                // Try fallback port
                port = 8766
                serverSocket = try { ServerSocket(port) } catch (_: Exception) { null }
            }

            Log.d("PairingServer", "Started on $pairingUrl")
            while (isRunning && serverSocket != null && !serverSocket!!.isClosed) {
                try {
                    val clientSocket = serverSocket!!.accept()
                    launch { handleClient(clientSocket) }
                } catch (e: Exception) {
                    if (!isRunning) break
                }
            }
        }

        // 2. Start cloud relay polling via ntfy.sh (fallback if phone is not on local Wi-Fi)
        pollJob = scope.launch {
            val topicUrl = "https://ntfy.sh/replayer_pair_$sessionId/raw?poll=1"
            var attempts = 0
            while (isActive && isRunning && attempts < 120) {
                delay(3000)
                attempts++
                try {
                    val request = Request.Builder().url(topicUrl).build()
                    val response = okClient.newCall(request).execute()
                    val body = response.body?.string()?.trim() ?: continue
                    if (body.isNotBlank() && body.contains("{") && body.contains("}")) {
                        Log.d("PairingServer", "Received ntfy payload: $body")
                        handleIncomingTokensJson(body)
                        break
                    }
                } catch (_: Exception) {}
            }
        }
    }

    fun stop() {
        isRunning = false
        pollJob?.cancel()
        try {
            serverSocket?.close()
        } catch (_: Exception) {}
        serverSocket = null
    }

    private suspend fun handleClient(socket: Socket) = withContext(Dispatchers.IO) {
        try {
            val reader = BufferedReader(InputStreamReader(socket.getInputStream()))
            val out = socket.getOutputStream()

            val requestLine = reader.readLine() ?: return@withContext
            val parts = requestLine.split(" ")
            if (parts.size < 2) return@withContext

            val method = parts[0]
            val path = parts[1]

            // Read headers to determine Content-Length
            var contentLength = 0
            var line: String?
            while (reader.readLine().also { line = it } != null) {
                if (line!!.isEmpty()) break
                if (line!!.lowercase().startsWith("content-length:")) {
                    contentLength = line!!.substringAfter(":").trim().toIntOrNull() ?: 0
                }
            }

            if (method.equals("POST", ignoreCase = true) && path.startsWith("/api/save")) {
                val bodyChars = CharArray(contentLength)
                var readTotal = 0
                while (readTotal < contentLength) {
                    val read = reader.read(bodyChars, readTotal, contentLength - readTotal)
                    if (read == -1) break
                    readTotal += read
                }
                val body = String(bodyChars)
                val success = handleIncomingPayload(body)
                sendHttpResponse(
                    out,
                    if (success) 200 else 400,
                    "application/json",
                    if (success) """{"status":"ok","message":"Google Drive успешно подключен на ТВ!"}"""
                    else """{"status":"error","message":"Не удалось сохранить токен"}"""
                )
            } else {
                // Serve mobile pairing web page
                val html = getPairingHtml(sessionId, authManager.getSavedClientId(), authManager.getSavedClientSecret())
                sendHttpResponse(out, 200, "text/html; charset=utf-8", html)
            }
        } catch (e: Exception) {
            Log.w("PairingServer", "Client error: ${e.message}")
        } finally {
            try { socket.close() } catch (_: Exception) {}
        }
    }

    private fun handleIncomingPayload(rawBody: String): Boolean {
        return try {
            if (rawBody.trim().startsWith("{")) {
                handleIncomingTokensJson(rawBody)
            } else {
                // URL-encoded form data
                val params = rawBody.split("&").associate { param ->
                    val pair = param.split("=")
                    val key = URLDecoder.decode(pair[0], "UTF-8")
                    val value = if (pair.size > 1) URLDecoder.decode(pair[1], "UTF-8") else ""
                    key to value
                }
                val clientId = params["client_id"]?.trim() ?: ""
                val clientSecret = params["client_secret"]?.trim() ?: ""
                val refreshToken = params["refresh_token"]?.trim() ?: ""
                val accessToken = params["access_token"]?.trim() ?: ""
                val authCode = params["auth_code"]?.trim() ?: ""

                if (clientId.isNotBlank()) authManager.saveClientId(clientId)
                if (clientSecret.isNotBlank()) authManager.saveClientSecret(clientSecret)

                if (authCode.isNotBlank()) {
                    scope.launch {
                        val result = authManager.exchangeAuthCode(authCode, clientId, clientSecret)
                        if (result.isSuccess) {
                            withContext(Dispatchers.Main) {
                                onPaired(authManager.getSavedAccountEmail())
                            }
                        }
                    }
                    return true
                }

                if (accessToken.isNotBlank() || refreshToken.isNotBlank()) {
                    authManager.saveTokens(
                        accessToken = accessToken.ifBlank { "valid_via_refresh" },
                        refreshToken = refreshToken.ifBlank { null },
                        expiresInSec = 3600
                    )
                    scope.launch {
                        if (refreshToken.isNotBlank()) {
                            authManager.refreshAccessToken()
                        }
                        withContext(Dispatchers.Main) {
                            onPaired(authManager.getSavedAccountEmail())
                        }
                    }
                    return true
                }
                false
            }
        } catch (e: Exception) {
            Log.e("PairingServer", "Failed to parse payload: ${e.message}")
            false
        }
    }

    private fun handleIncomingTokensJson(jsonStr: String): Boolean {
        return try {
            val element = json.parseToJsonElement(jsonStr) as? JsonObject ?: return false
            val clientId = element["client_id"]?.jsonPrimitive?.contentOrNull?.trim()
            val clientSecret = element["client_secret"]?.jsonPrimitive?.contentOrNull?.trim()
            val accessToken = element["access_token"]?.jsonPrimitive?.contentOrNull?.trim()
            val refreshToken = element["refresh_token"]?.jsonPrimitive?.contentOrNull?.trim()
            val authCode = element["auth_code"]?.jsonPrimitive?.contentOrNull?.trim()

            if (!clientId.isNullOrBlank()) authManager.saveClientId(clientId)
            if (!clientSecret.isNullOrBlank()) authManager.saveClientSecret(clientSecret)

            if (!authCode.isNullOrBlank()) {
                scope.launch {
                    val result = authManager.exchangeAuthCode(authCode, clientId ?: authManager.getSavedClientId(), clientSecret ?: authManager.getSavedClientSecret())
                    if (result.isSuccess) {
                        withContext(Dispatchers.Main) {
                            onPaired(authManager.getSavedAccountEmail())
                        }
                    }
                }
                return true
            }

            if (!accessToken.isNullOrBlank() || !refreshToken.isNullOrBlank()) {
                authManager.saveTokens(
                    accessToken = accessToken ?: "valid_via_refresh",
                    refreshToken = refreshToken,
                    expiresInSec = 3600
                )
                scope.launch {
                    if (!refreshToken.isNullOrBlank()) {
                        authManager.refreshAccessToken()
                    }
                    withContext(Dispatchers.Main) {
                        onPaired(authManager.getSavedAccountEmail())
                    }
                }
                return true
            }
            false
        } catch (e: Exception) {
            Log.e("PairingServer", "handleIncomingTokensJson error: ${e.message}")
            false
        }
    }

    private fun sendHttpResponse(out: OutputStream, code: Int, contentType: String, content: String) {
        val bytes = content.toByteArray(Charsets.UTF_8)
        val statusText = if (code == 200) "OK" else "Bad Request"
        val header = "HTTP/1.1 $code $statusText\r\n" +
                "Content-Type: $contentType\r\n" +
                "Content-Length: ${bytes.size}\r\n" +
                "Access-Control-Allow-Origin: *\r\n" +
                "Access-Control-Allow-Methods: GET, POST, OPTIONS\r\n" +
                "Access-Control-Allow-Headers: Content-Type\r\n" +
                "Connection: close\r\n\r\n"
        out.write(header.toByteArray(Charsets.UTF_8))
        out.write(bytes)
        out.flush()
    }

    private fun getLocalIpAddress(): String {
        try {
            val interfaces = NetworkInterface.getNetworkInterfaces()
            for (intf in interfaces) {
                if (intf.isLoopback || !intf.isUp) continue
                for (addr in intf.inetAddresses) {
                    if (addr is Inet4Address && !addr.isLoopbackAddress) {
                        val host = addr.hostAddress
                        if (host != null && (host.startsWith("192.168.") || host.startsWith("10.") || host.startsWith("172."))) {
                            return host
                        }
                    }
                }
            }
        } catch (_: Exception) {}
        return "127.0.0.1"
    }

@OptIn(UnstableApi::class)
class RePlayerEngine(private val context: Context) {

    private val trackSelector = DefaultTrackSelector(context)

    private val renderersFactory = DefaultRenderersFactory(context)
        .setExtensionRendererMode(DefaultRenderersFactory.EXTENSION_RENDERER_MODE_PREFER)
        .setEnableDecoderFallback(true)
        .setMediaCodecSelector { mimeType, requiresSecureDecoder, requiresTunnelingDecoder ->
            val decoders = MediaCodecSelector.DEFAULT.getDecoderInfos(mimeType, requiresSecureDecoder, requiresTunnelingDecoder)
            if (mimeType == MimeTypes.VIDEO_MP4V) {
                // Prioritize software decoders for MPEG4 / XviD / DivX as TV hardware decoders often have severe quirks
                decoders.sortedBy {
                    val isSoftware = it.softwareOnly ||
                            it.name.contains("google", ignoreCase = true) ||
                            it.name.contains("android", ignoreCase = true) ||
                            it.name.contains("sw", ignoreCase = true)
                    if (isSoftware) 0 else 1
                }
            } else {
                decoders
            }
        }

    private val loadControl = DefaultLoadControl.Builder()
        .setBufferDurationsMs(
            /* minBufferMs= */ 15_000,
            /* maxBufferMs= */ 50_000,
            /* bufferForPlaybackMs= */ 500,
            /* bufferForPlaybackAfterRebufferMs= */ 1_000
        )
        .setPrioritizeTimeOverSizeThresholds(false)
        .build()

    private val extractorsFactory = ExtractorsFactory {
        arrayOf(
            com.replayer.player.avi.AviExtractor(),
            *DefaultExtractorsFactory().setConstantBitrateSeekingEnabled(false).createExtractors()
        )
    }

    private val baseHttpDataSourceFactory = DefaultHttpDataSource.Factory()
        .setAllowCrossProtocolRedirects(true)
        .setConnectTimeoutMs(15000)
        .setReadTimeoutMs(30000)
        .setUserAgent("RePlayer-TV")

    private val defaultDataSourceFactory = DefaultDataSource.Factory(context, baseHttpDataSourceFactory)
    private val defaultMediaSourceFactory = DefaultMediaSourceFactory(defaultDataSourceFactory, extractorsFactory)

    val exoPlayer: ExoPlayer = ExoPlayer.Builder(context, renderersFactory)
        .setTrackSelector(trackSelector)
        .setLoadControl(loadControl)
        .setMediaSourceFactory(defaultMediaSourceFactory)
        .setSeekBackIncrementMs(10000)
        .setSeekForwardIncrementMs(10000)
        .setAudioAttributes(
            AudioAttributes.Builder()
                .setUsage(C.USAGE_MEDIA)
                .setContentType(C.AUDIO_CONTENT_TYPE_MOVIE)
                .build(),
            true
        )
        .build().apply {
            volume = 1.0f
        }

    private val _playerState = MutableStateFlow(RePlayerState())
    val playerState: StateFlow<RePlayerState> = _playerState.asStateFlow()

    private val scope = CoroutineScope(Dispatchers.Main)
    private var progressJob: Job? = null

    init {
        exoPlayer.addListener(object : Player.Listener {
            override fun onIsPlayingChanged(isPlaying: Boolean) {
                _playerState.value = _playerState.value.copy(isPlaying = isPlaying)
                if (isPlaying) startProgressTracking() else stopProgressTracking()
            }

            override fun onPlaybackStateChanged(playbackState: Int) {
                val isBuffering = playbackState == Player.STATE_BUFFERING
                val exoDuration = exoPlayer.duration
                val duration = if (exoDuration > 0) {
                    exoDuration
                } else {
                    _playerState.value.durationMs
                }
                _playerState.value = _playerState.value.copy(
                    isBuffering = isBuffering,
                    durationMs = duration
                )
            }

            override fun onTracksChanged(tracks: Tracks) {
                extractTracks(tracks)
            }

            override fun onPlayerError(error: PlaybackException) {
                _playerState.value = _playerState.value.copy(
                    errorMessage = error.localizedMessage ?: "Playback error",
                    isPlaying = false,
                    isBuffering = false
                )
            }
        })
    }

    fun play(mediaItem: MediaItem, headers: Map<String, String> = emptyMap()) {
        val mediaBuilder = ExoMediaItem.Builder()
            .setUri(mediaItem.uri)
            .setMediaId(mediaItem.id)

        val isAvi = mediaItem.title.endsWith(".avi", ignoreCase = true) ||
                mediaItem.uri.endsWith(".avi", ignoreCase = true) ||
                mediaItem.format?.equals("avi", ignoreCase = true) == true ||
                mediaItem.mimeType?.contains("avi", ignoreCase = true) == true ||
                mediaItem.mimeType?.contains("msvideo", ignoreCase = true) == true

        // Ensure container MIME type is explicitly specified (crucial for AVI, Google Drive, and DLNA streams)
        val detectedMimeType = when {
            isAvi -> MimeTypes.VIDEO_AVI
            mediaItem.title.endsWith(".mkv", ignoreCase = true) || mediaItem.uri.endsWith(".mkv", ignoreCase = true) -> MimeTypes.VIDEO_MATROSKA
            mediaItem.title.endsWith(".mp4", ignoreCase = true) || mediaItem.uri.endsWith(".mp4", ignoreCase = true) -> MimeTypes.VIDEO_MP4
            mediaItem.title.endsWith(".mov", ignoreCase = true) || mediaItem.uri.endsWith(".mov", ignoreCase = true) -> "video/quicktime"
            mediaItem.title.endsWith(".ts", ignoreCase = true) || mediaItem.uri.endsWith(".ts", ignoreCase = true) -> MimeTypes.VIDEO_MP2T
            mediaItem.title.endsWith(".mp3", ignoreCase = true) || mediaItem.uri.endsWith(".mp3", ignoreCase = true) -> MimeTypes.AUDIO_MPEG
            mediaItem.mimeType?.isNotBlank() == true && mediaItem.mimeType != "application/octet-stream" -> mediaItem.mimeType
            else -> null
        }
        if (detectedMimeType != null) {
            mediaBuilder.setMimeType(detectedMimeType)
        }

        _playerState.value = _playerState.value.copy(
            currentMediaTitle = mediaItem.title,
            errorMessage = null
        )

        val finalHeaders = headers.toMutableMap()
        if (mediaItem.sourceType == MediaSourceType.GOOGLE_DRIVE) {
            val authMgr = GoogleDriveAuthManager(context)
            val currentAccessToken = authMgr.getValidAccessTokenBlocking() ?: authMgr.getSavedAccessToken()
            if (!currentAccessToken.isNullOrBlank()) {
                finalHeaders["Authorization"] = "Bearer $currentAccessToken"
            }
        }

        val activeExtractorsFactory = object : ExtractorsFactory {
            override fun createExtractors(): Array<Extractor> {
                return arrayOf(
                    com.replayer.player.avi.AviExtractor(),
                    *DefaultExtractorsFactory().setConstantBitrateSeekingEnabled(false).createExtractors()
                )
            }

            override fun createExtractors(uri: Uri, responseHeaders: Map<String, List<String>>): Array<Extractor> {
                val cleanHeaders = if (isAvi) {
                    responseHeaders.filterKeys { !it.equals("Content-Type", ignoreCase = true) }
                } else {
                    responseHeaders
                }
                val defaultExtractors = DefaultExtractorsFactory()
                    .setConstantBitrateSeekingEnabled(false)
                    .createExtractors(uri, cleanHeaders)
                return arrayOf(com.replayer.player.avi.AviExtractor(), *defaultExtractors)
            }
        }

        val exoMediaItem = mediaBuilder.build()
        val httpDataSourceFactory = DefaultHttpDataSource.Factory()
            .setAllowCrossProtocolRedirects(true)
            .setConnectTimeoutMs(15000)
            .setReadTimeoutMs(30000)
            .setUserAgent("RePlayer-TV")

        if (finalHeaders.isNotEmpty()) {
            httpDataSourceFactory.setDefaultRequestProperties(finalHeaders)
        }

        val dataSourceFactory = DefaultDataSource.Factory(context, httpDataSourceFactory)
        val mediaSourceFactory = DefaultMediaSourceFactory(dataSourceFactory, activeExtractorsFactory)
        exoPlayer.setMediaSource(mediaSourceFactory.createMediaSource(exoMediaItem))

        exoPlayer.prepare()
        exoPlayer.playWhenReady = true
    }

    fun togglePlayPause() {
        if (exoPlayer.isPlaying) {
            exoPlayer.pause()
        } else {
            exoPlayer.play()
        }
    }

    fun seekRelative(deltaMs: Long) {
        val duration = if (exoPlayer.duration > 0) exoPlayer.duration else _playerState.value.durationMs
        val target = (exoPlayer.currentPosition + deltaMs).coerceIn(0, duration.coerceAtLeast(0L))
        exoPlayer.seekTo(target)
        _playerState.value = _playerState.value.copy(currentPositionMs = target)
    }

    fun seekTo(positionMs: Long) {
        val duration = if (exoPlayer.duration > 0) exoPlayer.duration else _playerState.value.durationMs
        val target = positionMs.coerceIn(0, duration.coerceAtLeast(0L))
        exoPlayer.seekTo(target)
        _playerState.value = _playerState.value.copy(currentPositionMs = target)
    }

    fun selectAudioTrack(track: AudioTrackInfo) {
        val tracks = exoPlayer.currentTracks
        for (groupIndex in 0 until tracks.groups.size) {
            val group = tracks.groups[groupIndex]
            if (group.type == C.TRACK_TYPE_AUDIO && groupIndex == track.groupIndex) {
                val trackGroup = group.mediaTrackGroup
                exoPlayer.trackSelectionParameters = exoPlayer.trackSelectionParameters.buildUpon()
                    .setOverrideForType(TrackSelectionOverride(trackGroup, listOf(track.index)))
                    .build()
                _playerState.value = _playerState.value.copy(selectedAudioTrackId = track.id)
                break
            }
        }
    }

    fun selectSubtitleTrack(track: SubtitleTrackInfo?) {
        if (track == null) {
            // Disable subtitles
            exoPlayer.trackSelectionParameters = exoPlayer.trackSelectionParameters.buildUpon()
                .setTrackTypeDisabled(C.TRACK_TYPE_TEXT, true)
                .build()
            _playerState.value = _playerState.value.copy(selectedSubtitleTrackId = null)
        } else {
            val tracks = exoPlayer.currentTracks
            for (groupIndex in 0 until tracks.groups.size) {
                val group = tracks.groups[groupIndex]
                if (group.type == C.TRACK_TYPE_TEXT && groupIndex == track.groupIndex) {
                    val trackGroup = group.mediaTrackGroup
                    exoPlayer.trackSelectionParameters = exoPlayer.trackSelectionParameters.buildUpon()
                        .setTrackTypeDisabled(C.TRACK_TYPE_TEXT, false)
                        .setOverrideForType(TrackSelectionOverride(trackGroup, listOf(track.index)))
                        .build()
                    _playerState.value = _playerState.value.copy(selectedSubtitleTrackId = track.id)
                    break
                }
            }
        }
    }

    private fun extractTracks(tracks: Tracks) {
        val audioList = mutableListOf<AudioTrackInfo>()
        val subList = mutableListOf<SubtitleTrackInfo>()
        var selectedAudioId: String? = null
        var selectedSubId: String? = null

        for (gIdx in 0 until tracks.groups.size) {
            val group = tracks.groups[gIdx]
            val trackGroup = group.mediaTrackGroup

            when (group.type) {
                C.TRACK_TYPE_AUDIO -> {
                    for (tIdx in 0 until group.length) {
                        val format = trackGroup.getFormat(tIdx)
                        val id = "audio_${gIdx}_${tIdx}"
                        val isSelected = group.isTrackSelected(tIdx)
                        if (isSelected) selectedAudioId = id

                        val lang = format.language ?: "und"
                        val channels = format.channelCount
                        val codec = format.sampleMimeType?.substringAfter("/")?.uppercase() ?: "AUDIO"
                        val label = format.label ?: "Track ${tIdx + 1} ($lang $channels.ch $codec)"

                        audioList.add(
                            AudioTrackInfo(
                                id = id,
                                index = tIdx,
                                groupIndex = gIdx,
                                language = format.language,
                                label = label,
                                codec = codec,
                                channelCount = channels,
                                sampleRate = format.sampleRate,
                                isSelected = isSelected
                            )
                        )
                    }
                }
                C.TRACK_TYPE_TEXT -> {
                    for (tIdx in 0 until group.length) {
                        val format = trackGroup.getFormat(tIdx)
                        val id = "sub_${gIdx}_${tIdx}"
                        val isSelected = group.isTrackSelected(tIdx)
                        if (isSelected) selectedSubId = id

                        val lang = format.language ?: "und"
                        val label = format.label ?: "Subtitle ${tIdx + 1} ($lang)"

                        subList.add(
                            SubtitleTrackInfo(
                                id = id,
                                index = tIdx,
                                groupIndex = gIdx,
                                language = format.language,
                                label = label,
                                mimeType = format.sampleMimeType,
                                isSelected = isSelected
                            )
                        )
                    }
                }
            }
        }

        _playerState.value = _playerState.value.copy(
            audioTracks = audioList,
            subtitleTracks = subList,
            selectedAudioTrackId = selectedAudioId,
            selectedSubtitleTrackId = selectedSubId
        )
    }

    private fun startProgressTracking() {
        progressJob?.cancel()
        progressJob = scope.launch {
            while (isActive) {
                _playerState.value = _playerState.value.copy(
                    currentPositionMs = exoPlayer.currentPosition.coerceAtLeast(0L),
                    durationMs = exoPlayer.duration.coerceAtLeast(0L)
                )
                delay(500)
            }
        }
    }

    private fun stopProgressTracking() {
        progressJob?.cancel()
        progressJob = null
    }

    fun release() {
        stopProgressTracking()
        exoPlayer.release()
    }
}

/**
 * Extracts data from the AVI container format.
 *
 * <p>Spec: https://docs.microsoft.com/en-us/windows/win32/directshow/avi-riff-file-reference.
 */
@UnstableApi
public final class AviExtractor implements Extractor {

  private static final String TAG = "AviExtractor";

  public static final int FOURCC_RIFF = 0x46464952;
  public static final int FOURCC_AVI_ = 0x20495641; // AVI<space>
  public static final int FOURCC_LIST = 0x5453494c;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_avih = 0x68697661;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_hdrl = 0x6c726468;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_strl = 0x6c727473;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_movi = 0x69766f6d;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_idx1 = 0x31786469;

  public static final int FOURCC_JUNK = 0x4b4e554a;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_strf = 0x66727473;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_strn = 0x6e727473;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_strh = 0x68727473;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_auds = 0x73647561;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_txts = 0x73747874;

  @SuppressWarnings("ConstantCaseForConstants")
  public static final int FOURCC_vids = 0x73646976;

  /** Parser states. */
  @Documented
  @Retention(RetentionPolicy.SOURCE)
  @Target(TYPE_USE)
  @IntDef({
    STATE_SKIPPING_TO_HDRL,
    STATE_READING_HDRL_HEADER,
    STATE_READING_HDRL_BODY,
    STATE_FINDING_MOVI_HEADER,
    STATE_FINDING_IDX1_HEADER,
    STATE_READING_IDX1_BODY,
    STATE_READING_SAMPLES,
  })
  private @interface State {}

  private static final int STATE_SKIPPING_TO_HDRL = 0;
  private static final int STATE_READING_HDRL_HEADER = 1;
  private static final int STATE_READING_HDRL_BODY = 2;
  private static final int STATE_FINDING_MOVI_HEADER = 3;
  private static final int STATE_FINDING_IDX1_HEADER = 4;
  private static final int STATE_READING_IDX1_BODY = 5;
  private static final int STATE_READING_SAMPLES = 6;

  private static final int AVIIF_KEYFRAME = 16;

  /**
   * Flags controlling the behavior of the extractor. Possible flag value is {@link
   * #FLAG_EMIT_RAW_SUBTITLE_DATA}.
   */
  @Documented
  @Retention(RetentionPolicy.SOURCE)
  @Target(TYPE_USE)
  @IntDef(
      flag = true,
      value = {FLAG_EMIT_RAW_SUBTITLE_DATA})
  public @interface Flags {}

  /**
   * Flag to use the source subtitle formats without modification. If unset, subtitles will be
   * transcoded to {@link MimeTypes#APPLICATION_MEDIA3_CUES} during extraction.
   */
  public static final int FLAG_EMIT_RAW_SUBTITLE_DATA = 1;

  /**
   * Maximum size to skip using {@link ExtractorInput#skip}. Boxes larger than this size are skipped
   * using {@link #RESULT_SEEK}.
   */
  private static final long RELOAD_MINIMUM_SEEK_DISTANCE = 256 * 1024;

  private final ParsableByteArray scratch;
  private final ChunkHeaderHolder chunkHeaderHolder;
  private final boolean parseSubtitlesDuringExtraction;
  private final SubtitleParser.Factory subtitleParserFactory;

  private @State int state;
  private ExtractorOutput extractorOutput;
  private @Nullable AviMainHeaderChunk aviHeader;
  private long durationUs;
  private ChunkReader[] chunkReaders;

  private long pendingReposition;
  private @Nullable ChunkReader currentChunkReader;
  private int hdrlSize;
  private long moviStart;
  private long moviEnd;
  private int idx1BodySize;
  private boolean seekMapHasBeenOutput;

  /**
   * @deprecated Use {@link #AviExtractor(int, SubtitleParser.Factory)} instead.
   */
  @Deprecated
  public AviExtractor() {
    this(FLAG_EMIT_RAW_SUBTITLE_DATA, SubtitleParser.Factory.UNSUPPORTED);
  }

  /**
   * Constructs an instance.
   *
   * @param extractorFlags Flags that control the extractor's behavior.
   * @param subtitleParserFactory The {@link SubtitleParser.Factory} for parsing subtitles during
   *     extraction.
   */
  public AviExtractor(@Flags int extractorFlags, SubtitleParser.Factory subtitleParserFactory) {
    this.subtitleParserFactory = subtitleParserFactory;
    parseSubtitlesDuringExtraction = (extractorFlags & FLAG_EMIT_RAW_SUBTITLE_DATA) == 0;
    scratch = new ParsableByteArray(/* limit= */ 12);
    chunkHeaderHolder = new ChunkHeaderHolder();
    extractorOutput = new NoOpExtractorOutput();
    chunkReaders = new ChunkReader[0];
    moviStart = C.INDEX_UNSET;
    moviEnd = C.INDEX_UNSET;
    hdrlSize = C.LENGTH_UNSET;
    durationUs = C.TIME_UNSET;
  }

  // Extractor implementation.

  @Override
  public void init(ExtractorOutput output) {
    this.state = STATE_SKIPPING_TO_HDRL;
    this.extractorOutput =
        parseSubtitlesDuringExtraction
            ? new SubtitleTranscodingExtractorOutput(output, subtitleParserFactory)
            : output;
    pendingReposition = C.INDEX_UNSET;
  }

  @Override
  public boolean sniff(ExtractorInput input) throws IOException {
    input.peekFully(scratch.getData(), /* offset= */ 0, /* length= */ 12);
    scratch.setPosition(0);
    if (scratch.readLittleEndianInt() != FOURCC_RIFF) {
      return false;
    }
    scratch.skipBytes(4); // Skip the RIFF chunk length.
    return scratch.readLittleEndianInt() == FOURCC_AVI_;
  }

  @Override
  public int read(ExtractorInput input, PositionHolder seekPosition) throws IOException {
    if (resolvePendingReposition(input, seekPosition)) {
      return RESULT_SEEK;
    }
    switch (state) {
      case STATE_SKIPPING_TO_HDRL:
        // Check for RIFF and AVI fourcc's just in case the caller did not sniff, in order to
        // provide a meaningful error if the input is not an AVI file.
        if (sniff(input)) {
          input.skipFully(/* length= */ 12);
        } else {
          throw ParserException.createForMalformedContainer(
              /* message= */ "AVI Header List not found", /* cause= */ null);
        }
        state = STATE_READING_HDRL_HEADER;
        return RESULT_CONTINUE;
      case STATE_READING_HDRL_HEADER:
        input.readFully(scratch.getData(), /* offset= */ 0, /* length= */ 12);
        scratch.setPosition(0);
        chunkHeaderHolder.populateWithListHeaderFrom(scratch);
        if (chunkHeaderHolder.listType != FOURCC_hdrl) {
          throw ParserException.createForMalformedContainer(
              /* message= */ "hdrl expected, found: " + chunkHeaderHolder.listType,
              /* cause= */ null);
        }
        hdrlSize = chunkHeaderHolder.size;
        state = STATE_READING_HDRL_BODY;
        return RESULT_CONTINUE;
      case STATE_READING_HDRL_BODY:
        // hdrlSize includes the LIST type (hdrl), so we subtract 4 to the size.
        int bytesToRead = hdrlSize - 4;
        ParsableByteArray hdrlBody = new ParsableByteArray(bytesToRead);
        input.readFully(hdrlBody.getData(), /* offset= */ 0, bytesToRead);
        parseHdrlBody(hdrlBody);
        state = STATE_FINDING_MOVI_HEADER;
        return RESULT_CONTINUE;
      case STATE_FINDING_MOVI_HEADER:
        if (moviStart != C.INDEX_UNSET && input.getPosition() != moviStart) {
          pendingReposition = moviStart;
          return RESULT_CONTINUE;
        }
        input.peekFully(scratch.getData(), /* offset= */ 0, /* length= */ 12);
        input.resetPeekPosition();
        scratch.setPosition(0);
        chunkHeaderHolder.populateFrom(scratch);
        int listType = scratch.readLittleEndianInt();
        if (chunkHeaderHolder.chunkType == FOURCC_RIFF) {
          // We are at the start of the file. The movi chunk is in the RIFF chunk, so we skip the
          // header, so as to read the RIFF chunk's body.
          input.skipFully(12);
          return RESULT_CONTINUE;
        }
        if (chunkHeaderHolder.chunkType != FOURCC_LIST || listType != FOURCC_movi) {
          // The chunk header (8 bytes) plus the whole body.
          pendingReposition = input.getPosition() + chunkHeaderHolder.size + 8;
          return RESULT_CONTINUE;
        }
        moviStart = input.getPosition();
        // Size includes the list type, but not the LIST or size fields, so we add 8.
        moviEnd = moviStart + chunkHeaderHolder.size + 8;
        if (!seekMapHasBeenOutput) {
          if (checkNotNull(aviHeader).hasIndex()) {
            state = STATE_FINDING_IDX1_HEADER;
            pendingReposition = moviEnd;
            return RESULT_CONTINUE;
          } else {
            extractorOutput.seekMap(new SeekMap.Unseekable(durationUs));
            seekMapHasBeenOutput = true;
          }
        }
        // No need to parse the idx1, so we start reading the samples from the movi chunk straight
        // away. We skip 12 bytes to move to the start of the movi's body.
        pendingReposition = input.getPosition() + 12;
        state = STATE_READING_SAMPLES;
        return RESULT_CONTINUE;
      case STATE_FINDING_IDX1_HEADER:
        input.readFully(scratch.getData(), /* offset= */ 0, /* length= */ 8);
        scratch.setPosition(0);
        int idx1Fourcc = scratch.readLittleEndianInt();
        int boxSize = scratch.readLittleEndianInt();
        if (idx1Fourcc == FOURCC_idx1) {
          state = STATE_READING_IDX1_BODY;
          idx1BodySize = boxSize;
        } else {
          // This one is not idx1, skip to the next box.
          pendingReposition = input.getPosition() + boxSize;
        }
        return RESULT_CONTINUE;
      case STATE_READING_IDX1_BODY:
        ParsableByteArray idx1Body = new ParsableByteArray(idx1BodySize);
        input.readFully(idx1Body.getData(), /* offset= */ 0, /* length= */ idx1BodySize);
        parseIdx1Body(idx1Body);
        state = STATE_READING_SAMPLES;
        pendingReposition = moviStart;
        return RESULT_CONTINUE;
      case STATE_READING_SAMPLES:
        return readMoviChunks(input);
      default:
        throw new AssertionError(); // Should never happen.
    }
  }

  @Override
  public void seek(long position, long timeUs) {
    pendingReposition = C.INDEX_UNSET;
    currentChunkReader = null;
    for (ChunkReader chunkReader : chunkReaders) {
      chunkReader.seekToPosition(position);
    }
    if (position == 0) {
      if (chunkReaders.length == 0) {
        // Still unprepared.
        state = STATE_SKIPPING_TO_HDRL;
      } else {
        state = STATE_FINDING_MOVI_HEADER;
      }
      return;
    }
    state = STATE_READING_SAMPLES;
  }

  @Override
  public void release() {
    // Nothing to release.
  }

  // Internal methods.

  /**
   * Returns whether a {@link #RESULT_SEEK} is required for the pending reposition. A seek may not
   * be necessary when the desired position (as held by {@link #pendingReposition}) is after the
   * {@link ExtractorInput#getPosition() current position}, but not further than {@link
   * #RELOAD_MINIMUM_SEEK_DISTANCE}.
   */
  private boolean resolvePendingReposition(ExtractorInput input, PositionHolder seekPosition)
      throws IOException {
    boolean needSeek = false;
    if (pendingReposition != C.INDEX_UNSET) {
      long currentPosition = input.getPosition();
      if (pendingReposition < currentPosition
          || pendingReposition > currentPosition + RELOAD_MINIMUM_SEEK_DISTANCE) {
        seekPosition.position = pendingReposition;
        needSeek = true;
      } else {
        // The distance to the target position is short enough that it makes sense to just skip the
        // bytes, instead of doing a seek which might re-create an HTTP connection.
        input.skipFully((int) (pendingReposition - currentPosition));
      }
    }
    pendingReposition = C.INDEX_UNSET;
    return needSeek;
  }

  private void parseHdrlBody(ParsableByteArray hrdlBody) throws IOException {
    ListChunk headerList = ListChunk.parseFrom(FOURCC_hdrl, hrdlBody);
    if (headerList.getType() != FOURCC_hdrl) {
      throw ParserException.createForMalformedContainer(
          /* message= */ "Unexpected header list type " + headerList.getType(), /* cause= */ null);
    }
    @Nullable AviMainHeaderChunk aviHeader = headerList.getChild(AviMainHeaderChunk.class);
    if (aviHeader == null) {
      throw ParserException.createForMalformedContainer(
          /* message= */ "AviHeader not found", /* cause= */ null);
    }
    this.aviHeader = aviHeader;
    // This is usually wrong, so it will be overwritten by video if present
    durationUs = aviHeader.totalFrames * (long) aviHeader.frameDurationUs;
    ArrayList<ChunkReader> chunkReaderList = new ArrayList<>();
    int streamId = 0;
    for (AviChunk aviChunk : headerList.children) {
      if (aviChunk.getType() == FOURCC_strl) {
        ListChunk streamList = (ListChunk) aviChunk;
        // Note the streamId needs to increment even if the corresponding `strl` is discarded.
        // See
        // https://docs.microsoft.com/en-us/windows/win32/directshow/avi-riff-file-reference#avi-stream-headers.
        @Nullable ChunkReader chunkReader = processStreamList(streamList, streamId++);
        if (chunkReader != null) {
          chunkReaderList.add(chunkReader);
        }
      }
    }
    chunkReaders = chunkReaderList.toArray(new ChunkReader[0]);
    extractorOutput.endTracks();
  }

  /** Builds and outputs the {@link SeekMap} from the idx1 chunk. */
  private void parseIdx1Body(ParsableByteArray body) {
    long seekOffset = peekSeekOffset(body);
    while (body.bytesLeft() >= 16) {
      int chunkId = body.readLittleEndianInt();
      int flags = body.readLittleEndianInt();
      long offset = body.readLittleEndianInt() + seekOffset;
      body.skipBytes(4); // Ignore size.
      ChunkReader chunkReader = getChunkReader(chunkId);
      if (chunkReader == null) {
        // We ignore unknown chunk IDs.
        continue;
      }
      chunkReader.appendIndexChunk(
          offset, /* isKeyFrame= */ (flags & AVIIF_KEYFRAME) == AVIIF_KEYFRAME);
    }
    for (ChunkReader chunkReader : chunkReaders) {
      chunkReader.commitIndex();
    }
    seekMapHasBeenOutput = true;
    if (chunkReaders.length == 0) {
      extractorOutput.seekMap(new SeekMap.Unseekable(durationUs));
    } else {
      extractorOutput.seekMap(new AviSeekMap(durationUs));
    }
  }

  private long peekSeekOffset(ParsableByteArray idx1Body) {
    // The spec states the offset is based on the start of the movi list type fourcc, but it also
    // says some files base the offset on the start of the file. We use a best effort approach to
    // figure out which is the case. See:
    // https://docs.microsoft.com/en-us/previous-versions/windows/desktop/api/Aviriff/ns-aviriff-avioldindex#dwoffset.
    if (idx1Body.bytesLeft() < 16) {
      // There are no full entries in the index, meaning we don't need to apply an offset.
      return 0;
    }
    int startingPosition = idx1Body.getPosition();
    idx1Body.skipBytes(8); // Skip chunkId (4 bytes) and flags (4 bytes).
    int offset = idx1Body.readLittleEndianInt();

    // moviStart points at the start of the LIST, while the seek offset is based at the start of the
    // movi fourCC, so we add 8 to reconcile the difference.
    long seekOffset = offset > moviStart ? 0L : moviStart + 8;
    idx1Body.setPosition(startingPosition);
    return seekOffset;
  }

  private @Nullable ChunkReader getChunkReader(int chunkId) {
    for (ChunkReader chunkReader : chunkReaders) {
      if (chunkReader.handlesChunkId(chunkId)) {
        return chunkReader;
      }
    }
    return null;
  }

  private int readMoviChunks(ExtractorInput input) throws IOException {
    if (input.getPosition() >= moviEnd) {
      return C.RESULT_END_OF_INPUT;
    } else if (currentChunkReader != null) {
      if (currentChunkReader.onChunkData(input)) {
        currentChunkReader = null;
      }
    } else {
      alignInputToEvenPosition(input);
      input.peekFully(scratch.getData(), /* offset= */ 0, 12);
      scratch.setPosition(0);
      int chunkType = scratch.readLittleEndianInt();
      if (chunkType == FOURCC_LIST) {
        scratch.setPosition(8);
        int listType = scratch.readLittleEndianInt();
        input.skipFully(listType == FOURCC_movi ? 12 : 8);
        input.resetPeekPosition();
        return RESULT_CONTINUE;
      }
      int size = scratch.readLittleEndianInt();
      if (chunkType == FOURCC_JUNK) {
        pendingReposition = input.getPosition() + size + 8;
        return RESULT_CONTINUE;
      }
      input.skipFully(8);
      input.resetPeekPosition();
      ChunkReader chunkReader = getChunkReader(chunkType);
      if (chunkReader == null) {
        // No handler for this chunk. We skip it.
        pendingReposition = input.getPosition() + size;
        return RESULT_CONTINUE;
      } else {
        chunkReader.onChunkStart(size);
        this.currentChunkReader = chunkReader;
      }
    }
    return RESULT_CONTINUE;
  }

  private @Nullable ChunkReader processStreamList(ListChunk streamList, int streamId) {
    AviStreamHeaderChunk aviStreamHeaderChunk = streamList.getChild(AviStreamHeaderChunk.class);
    StreamFormatChunk streamFormatChunk = streamList.getChild(StreamFormatChunk.class);
    if (aviStreamHeaderChunk == null) {
      Log.w(TAG, "Missing Stream Header");
      return null;
    }
    if (streamFormatChunk == null) {
      Log.w(TAG, "Missing Stream Format");
      return null;
    }
    long durationUs = aviStreamHeaderChunk.getDurationUs();
    long movieDurationUs = aviHeader != null
        ? ((long) aviHeader.totalFrames * aviHeader.frameDurationUs)
        : durationUs;
    if (durationUs <= 0) {
      durationUs = movieDurationUs;
    }
    Format streamFormat = streamFormatChunk.format;
    Format.Builder builder = streamFormat.buildUpon();
    builder.setId(streamId);
    int suggestedBufferSize = aviStreamHeaderChunk.suggestedBufferSize;
    if (suggestedBufferSize != 0) {
      builder.setMaxInputSize(suggestedBufferSize);
    }
    StreamNameChunk streamName = streamList.getChild(StreamNameChunk.class);
    if (streamName != null) {
      builder.setLabel(streamName.name);
    }
    int trackType = MimeTypes.getTrackType(streamFormat.sampleMimeType);
    if (trackType == C.TRACK_TYPE_AUDIO || trackType == C.TRACK_TYPE_VIDEO) {
      TrackOutput trackOutput = extractorOutput.track(streamId, trackType);
      Format format = builder.build();
      trackOutput.format(format);
      this.durationUs = max(this.durationUs, durationUs);
      return new ChunkReader(
          streamId,
          aviStreamHeaderChunk,
          trackOutput,
          MimeTypes.allSamplesAreSyncSamples(format.sampleMimeType, format.codecs),
          movieDurationUs);
    } else {
      // We don't currently support tracks other than video and audio.
      return null;
    }
  }

  /**
   * Skips one byte from the given {@code input} if the current position is odd.
   *
   * <p>This isn't documented anywhere, but AVI files are aligned to even bytes and fill gaps with
   * zeros.
   */
  private static void alignInputToEvenPosition(ExtractorInput input) throws IOException {
    if ((input.getPosition() & 1) == 1) {
      input.skipFully(1);
    }
  }

  // Internal classes.

  private class AviSeekMap implements SeekMap {

    private final long durationUs;

    public AviSeekMap(long durationUs) {
      this.durationUs = durationUs;
    }

    @Override
    public boolean isSeekable() {
      return true;
    }

    @Override
    public long getDurationUs() {
      return durationUs;
    }

    @Override
    public SeekPoints getSeekPoints(long timeUs) {
      SeekPoints result = chunkReaders[0].getSeekPoints(timeUs);
      for (int i = 1; i < chunkReaders.length; i++) {
        SeekPoints seekPoints = chunkReaders[i].getSeekPoints(timeUs);
        if (seekPoints.first.position < result.first.position) {
          result = seekPoints;
        }
      }
      return result;
    }
  }

  private static class ChunkHeaderHolder {
    public int chunkType;
    public int size;
    public int listType;

    public void populateWithListHeaderFrom(ParsableByteArray headerBytes) throws ParserException {
      populateFrom(headerBytes);
      if (chunkType != AviExtractor.FOURCC_LIST) {
        throw ParserException.createForMalformedContainer(
            /* message= */ "LIST expected, found: " + chunkType, /* cause= */ null);
      }
      listType = headerBytes.readLittleEndianInt();
    }

    public void populateFrom(ParsableByteArray headerBytes) {
      chunkType = headerBytes.readLittleEndianInt();
      size = headerBytes.readLittleEndianInt();
      listType = 0;
    }
  }
}

import static androidx.media3.common.C.TRACK_TYPE_AUDIO;
import static androidx.media3.common.C.TRACK_TYPE_VIDEO;
import static com.google.common.base.Preconditions.checkArgument;
import static java.lang.annotation.ElementType.TYPE_USE;

import androidx.annotation.IntDef;
import androidx.media3.common.C;
import androidx.media3.common.util.Util;
import androidx.media3.extractor.ExtractorInput;
import androidx.media3.extractor.SeekMap;
import androidx.media3.extractor.SeekPoint;
import androidx.media3.extractor.TrackOutput;
import java.io.IOException;
import java.lang.annotation.Documented;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;
import java.util.Arrays;

/** Reads chunks holding sample data. */
/* package */ final class ChunkReader {

  /** Parser states. */
  @Documented
  @Retention(RetentionPolicy.SOURCE)
  @Target(TYPE_USE)
  @IntDef({
    CHUNK_TYPE_VIDEO_COMPRESSED,
    CHUNK_TYPE_VIDEO_UNCOMPRESSED,
    CHUNK_TYPE_AUDIO,
  })
  private @interface ChunkType {}

  private static final int INITIAL_INDEX_SIZE = 512;
  private static final int CHUNK_TYPE_VIDEO_COMPRESSED = ('d' << 16) | ('c' << 24);
  private static final int CHUNK_TYPE_VIDEO_UNCOMPRESSED = ('d' << 16) | ('b' << 24);
  private static final int CHUNK_TYPE_AUDIO = ('w' << 16) | ('b' << 24);

  private final AviStreamHeaderChunk streamHeaderChunk;
  private final TrackOutput trackOutput;

  /** The chunk id fourCC (example: `01wb`), as defined in the index and the movi. */
  private final int chunkId;

  /** Secondary chunk id. Bad muxers sometimes use an uncompressed video id (db) for key frames. */
  private final int alternativeChunkId;

  private final long durationUs;
  private final boolean allSamplesAreSyncSamples;

  private int chunkCount;
  private int currentChunkSize;
  private int bytesRemainingInCurrentChunk;

  /** Number of chunks as calculated by the index. */
  private int currentChunkIndex;

  private int indexChunkCount;
  private int indexSize;
  private long firstIndexChunkOffset;
  private long[] keyFrameOffsets;
  private int[] keyFrameIndices;

  public ChunkReader(
      int id,
      AviStreamHeaderChunk streamHeaderChunk,
      TrackOutput trackOutput,
      boolean allSamplesAreSyncSamples) {
    this(id, streamHeaderChunk, trackOutput, allSamplesAreSyncSamples, C.TIME_UNSET);
  }

  public ChunkReader(
      int id,
      AviStreamHeaderChunk streamHeaderChunk,
      TrackOutput trackOutput,
      boolean allSamplesAreSyncSamples,
      long movieDurationUs) {
    this.streamHeaderChunk = streamHeaderChunk;
    this.allSamplesAreSyncSamples = allSamplesAreSyncSamples;
    @C.TrackType int trackType = streamHeaderChunk.getTrackType();
    checkArgument(trackType == TRACK_TYPE_AUDIO || trackType == TRACK_TYPE_VIDEO);
    @ChunkType
    int chunkType = trackType == TRACK_TYPE_VIDEO ? CHUNK_TYPE_VIDEO_COMPRESSED : CHUNK_TYPE_AUDIO;
    chunkId = getChunkIdFourCc(id, chunkType);
    long headerDuration = streamHeaderChunk.getDurationUs();
    if (trackType == TRACK_TYPE_AUDIO) {
      if (movieDurationUs > 0 && (headerDuration <= 0 || Math.abs(headerDuration - movieDurationUs) > movieDurationUs / 2)) {
        durationUs = movieDurationUs;
      } else {
        durationUs = headerDuration > 0 ? headerDuration : (movieDurationUs > 0 ? movieDurationUs : C.TIME_UNSET);
      }
    } else {
      durationUs = headerDuration > 0 ? headerDuration : (movieDurationUs > 0 ? movieDurationUs : C.TIME_UNSET);
    }
    this.trackOutput = trackOutput;
    alternativeChunkId =
        trackType == TRACK_TYPE_VIDEO ? getChunkIdFourCc(id, CHUNK_TYPE_VIDEO_UNCOMPRESSED) : -1;
    firstIndexChunkOffset = C.INDEX_UNSET;
    keyFrameOffsets = new long[INITIAL_INDEX_SIZE];
    keyFrameIndices = new int[INITIAL_INDEX_SIZE];
    chunkCount = streamHeaderChunk.length > 0 ? streamHeaderChunk.length : 1;
  }

  public void appendIndexChunk(long offset, boolean isKeyFrame) {
    if (firstIndexChunkOffset == C.INDEX_UNSET) {
      firstIndexChunkOffset = offset;
    }
    if (isKeyFrame || allSamplesAreSyncSamples) {
      if (indexSize == keyFrameIndices.length) {
        keyFrameOffsets = Arrays.copyOf(keyFrameOffsets, keyFrameOffsets.length * 3 / 2);
        keyFrameIndices = Arrays.copyOf(keyFrameIndices, keyFrameIndices.length * 3 / 2);
      }
      keyFrameOffsets[indexSize] = offset;
      keyFrameIndices[indexSize] = indexChunkCount;
      indexSize++;
    }
    indexChunkCount++;
  }

  public void advanceCurrentChunk() {
    currentChunkIndex++;
  }

  public long getCurrentChunkTimestampUs() {
    return getChunkTimestampUs(currentChunkIndex);
  }

  public long getFrameDurationUs() {
    if (isVideo() && streamHeaderChunk.rate > 0 && streamHeaderChunk.scale > 0) {
      long duration = Util.scaleLargeTimestamp(
          1,
          (long) streamHeaderChunk.scale * C.MICROS_PER_SECOND,
          streamHeaderChunk.rate);
      return duration > 0 ? duration : 40_000L;
    }
    long count = chunkCount > 0 ? chunkCount : 1;
    long totalDuration = durationUs > 0 ? durationUs : 0;
    long chunkDur = count > 0 ? totalDuration / count : 40_000L;
    return chunkDur > 0 ? chunkDur : 40_000L;
  }

  public void commitIndex() {
    keyFrameOffsets = Arrays.copyOf(keyFrameOffsets, indexSize);
    keyFrameIndices = Arrays.copyOf(keyFrameIndices, indexSize);
    if (indexChunkCount > 0) {
      chunkCount = indexChunkCount;
    } else if (streamHeaderChunk.length > 0) {
      chunkCount = streamHeaderChunk.length;
    }
  }

  public boolean handlesChunkId(int chunkId) {
    return this.chunkId == chunkId || alternativeChunkId == chunkId;
  }

  public boolean isCurrentFrameAKeyFrame() {
    return allSamplesAreSyncSamples || Arrays.binarySearch(keyFrameIndices, currentChunkIndex) >= 0;
  }

  public boolean isVideo() {
    return streamHeaderChunk.getTrackType() == TRACK_TYPE_VIDEO;
  }

  public boolean isAudio() {
    return streamHeaderChunk.getTrackType() == TRACK_TYPE_AUDIO;
  }

  /** Prepares for parsing a chunk with the given {@code size}. */
  public void onChunkStart(int size) {
    currentChunkSize = size;
    bytesRemainingInCurrentChunk = size;
  }

  /**
   * Provides data associated to the current chunk and returns whether the full chunk has been
   * parsed.
   */
  public boolean onChunkData(ExtractorInput input) throws IOException {
    bytesRemainingInCurrentChunk -=
        trackOutput.sampleData(input, bytesRemainingInCurrentChunk, false);
    boolean done = bytesRemainingInCurrentChunk == 0;
    if (done) {
      if (currentChunkSize > 0) {
        trackOutput.sampleMetadata(
            getCurrentChunkTimestampUs(),
            (isCurrentFrameAKeyFrame() ? C.BUFFER_FLAG_KEY_FRAME : 0),
            currentChunkSize,
            0,
            null);
      }
      advanceCurrentChunk();
    }
    return done;
  }

  public void seekToPosition(long position) {
    if (indexSize == 0) {
      currentChunkIndex = 0;
    } else {
      int index =
          Util.binarySearchFloor(
              keyFrameOffsets, position, /* inclusive= */ true, /* stayInBounds= */ true);
      currentChunkIndex = keyFrameIndices[index];
    }
  }

  public SeekMap.SeekPoints getSeekPoints(long timeUs) {
    if (indexSize == 0) {
      // Return the offset of the first chunk as there are no keyframes in the index.
      return new SeekMap.SeekPoints(
          new SeekPoint(/* timeUs= */ 0, /* position= */ firstIndexChunkOffset));
    }
    long frameDur = getFrameDurationUs();
    int targetFrameIndex = (int) (timeUs / (frameDur > 0 ? frameDur : 40_000L));
    int keyFrameIndex =
        Util.binarySearchFloor(
            keyFrameIndices, targetFrameIndex, /* inclusive= */ true, /* stayInBounds= */ true);
    if (keyFrameIndices[keyFrameIndex] == targetFrameIndex) {
      return new SeekMap.SeekPoints(getSeekPoint(keyFrameIndex));
    }
    // The target frame is not a key frame, we look for the two closest ones.
    SeekPoint precedingKeyFrameSeekPoint = getSeekPoint(keyFrameIndex);
    if (keyFrameIndex + 1 < keyFrameOffsets.length) {
      return new SeekMap.SeekPoints(precedingKeyFrameSeekPoint, getSeekPoint(keyFrameIndex + 1));
    } else {
      return new SeekMap.SeekPoints(precedingKeyFrameSeekPoint);
    }
  }

  private long getChunkTimestampUs(int chunkIndex) {
    if (isVideo() && streamHeaderChunk.rate > 0 && streamHeaderChunk.scale > 0) {
      return Util.scaleLargeTimestamp(
          chunkIndex,
          (long) streamHeaderChunk.scale * C.MICROS_PER_SECOND,
          streamHeaderChunk.rate);
    }
    long count = chunkCount > 0 ? chunkCount : 1;
    long totalDuration = durationUs > 0 ? durationUs : 0;
    return totalDuration * chunkIndex / count;
  }

  private SeekPoint getSeekPoint(int keyFrameIndex) {
    return new SeekPoint(
        keyFrameIndices[keyFrameIndex] * getFrameDurationUs(), keyFrameOffsets[keyFrameIndex]);
  }

  private static int getChunkIdFourCc(int streamId, @ChunkType int chunkType) {
    int tens = streamId / 10;
    int ones = streamId % 10;
    return (('0' + ones) << 8) | ('0' + tens) | chunkType;
  }
}

/** Wrapper around the AVIMAINHEADER structure */
/* package */ final class AviMainHeaderChunk implements AviChunk {

  private static final int AVIF_HAS_INDEX = 0x10;

  public static AviMainHeaderChunk parseFrom(ParsableByteArray body) {
    int microSecPerFrame = body.readLittleEndianInt();

    this.flags = flags;
    this.totalFrames = totalFrames;
    this.streams = streams;
  }

  @Override
  public int getType() {
    return AviExtractor.FOURCC_avih;
  }

  public boolean hasIndex() {
    return (flags & AVIF_HAS_INDEX) == AVIF_HAS_INDEX;
  }

import androidx.media3.common.C;
import androidx.media3.common.util.Log;
import androidx.media3.common.util.ParsableByteArray;
import androidx.media3.common.util.Util;

/** Parses and holds information from the AVISTREAMHEADER structure. */
/* package */ final class AviStreamHeaderChunk implements AviChunk {
  private static final String TAG = "AviStreamHeaderChunk";

  public static AviStreamHeaderChunk parseFrom(ParsableByteArray body) {
    int streamType = body.readLittleEndianInt();
    body.skipBytes(12); // fccHandler (4 bytes), dwFlags (4 bytes), wPriority (2 bytes),
    // wLanguage (2 bytes).
    int initialFrames = body.readLittleEndianInt();
    int scale = body.readLittleEndianInt();
    int rate = body.readLittleEndianInt();
    body.skipBytes(4); // dwStart (4 bytes).
    int length = body.readLittleEndianInt();
    int suggestedBufferSize = body.readLittleEndianInt();
    body.skipBytes(4); // dwQuality (4 bytes).
    int sampleSize = body.readLittleEndianInt();
    return new AviStreamHeaderChunk(
        streamType, initialFrames, scale, rate, length, suggestedBufferSize, sampleSize);
  }

  public final int streamType;
  public final int initialFrames;
  public final int scale;
  public final int rate;
  public final int length;
  public final int suggestedBufferSize;
  public final int sampleSize;

  private AviStreamHeaderChunk(
      int streamType,
      int initialFrames,
      int scale,
      int rate,
      int length,
      int suggestedBufferSize,
      int sampleSize) {
    this.streamType = streamType;
    this.initialFrames = initialFrames;
    this.scale = scale;
    this.rate = rate;
    this.length = length;
    this.suggestedBufferSize = suggestedBufferSize;
    this.sampleSize = sampleSize;
  }

  @Override
  public int getType() {
    return AviExtractor.FOURCC_strh;
  }

  public @C.TrackType int getTrackType() {
    switch (streamType) {
      case AviExtractor.FOURCC_auds:
        return C.TRACK_TYPE_AUDIO;
      case AviExtractor.FOURCC_vids:
        return C.TRACK_TYPE_VIDEO;
      case AviExtractor.FOURCC_txts:
        return C.TRACK_TYPE_TEXT;
      default:
        Log.w(TAG, "Found unsupported streamType fourCC: " + Integer.toHexString(streamType));
        return C.TRACK_TYPE_UNKNOWN;
    }
  }

  public float getFrameRate() {
    return rate / (float) scale;
  }

  public long getDurationUs() {
    return Util.scaleLargeTimestamp(
        /* timestamp= */ length,
        /* multiplier= */ C.MICROS_PER_SECOND * scale,
        /* divisor= */ rate);
  }
}

package com.replayer.player.avi;

import androidx.media3.common.C;
import androidx.media3.common.util.ParsableByteArray;
import com.google.common.collect.ImmutableList;
import org.jspecify.annotations.Nullable;

/** Represents an AVI LIST. */
/* package */ final class ListChunk implements AviChunk {

  public static ListChunk parseFrom(int listType, ParsableByteArray body) {
    ImmutableList.Builder<AviChunk> builder = new ImmutableList.Builder<>();
    int listBodyEndPosition = body.limit();
    @C.TrackType int currentTrackType = C.TRACK_TYPE_NONE;
    while (body.bytesLeft() > 8) {
      int type = body.readLittleEndianInt();
      int size = body.readLittleEndianInt();
      int innerBoxBodyEndPosition = body.getPosition() + size;
      body.setLimit(innerBoxBodyEndPosition);
      @Nullable AviChunk aviChunk;
      if (type == AviExtractor.FOURCC_LIST) {
        int innerListType = body.readLittleEndianInt();
        aviChunk = parseFrom(innerListType, body);
      } else {
        aviChunk = createBox(type, currentTrackType, body);
      }
      if (aviChunk != null) {
        if (aviChunk.getType() == AviExtractor.FOURCC_strh) {
          currentTrackType = ((AviStreamHeaderChunk) aviChunk).getTrackType();
        }
        builder.add(aviChunk);
      }
      body.setPosition(innerBoxBodyEndPosition);
      body.setLimit(listBodyEndPosition);
    }
    return new ListChunk(listType, builder.build());
  }

  public final ImmutableList<AviChunk> children;
  private final int type;

  private ListChunk(int type, ImmutableList<AviChunk> children) {
    this.type = type;
    this.children = children;
  }

  @Override
  public int getType() {
    return type;
  }

  @SuppressWarnings("unchecked")
  public <T extends AviChunk> @Nullable T getChild(Class<T> c) {
    for (AviChunk aviChunk : children) {
      if (aviChunk.getClass() == c) {
        return (T) aviChunk;
      }
    }
    return null;
  }

  private static @Nullable AviChunk createBox(
      int chunkType, @C.TrackType int trackType, ParsableByteArray body) {
    switch (chunkType) {
      case AviExtractor.FOURCC_avih:
        return AviMainHeaderChunk.parseFrom(body);
      case AviExtractor.FOURCC_strh:
        return AviStreamHeaderChunk.parseFrom(body);
      case AviExtractor.FOURCC_strf:
        return StreamFormatChunk.parseFrom(trackType, body);
      case AviExtractor.FOURCC_strn:
        return StreamNameChunk.parseFrom(body);
      default:
        return null;
    }
  }
}

import androidx.media3.common.C;
import androidx.media3.common.Format;
import androidx.media3.common.MimeTypes;
import androidx.media3.common.util.Log;

import androidx.media3.common.util.Util;
import com.google.common.collect.ImmutableList;
import org.jspecify.annotations.Nullable;

/** Holds the {@link Format} information contained in an STRF chunk. */
/* package */ final class StreamFormatChunk implements AviChunk {
  private static final String TAG = "StreamFormatChunk";

  public static @Nullable AviChunk parseFrom(int trackType, ParsableByteArray body) {
    if (trackType == C.TRACK_TYPE_VIDEO) {
      return parseBitmapInfoHeader(body);
    } else if (trackType == C.TRACK_TYPE_AUDIO) {
      return parseWaveFormatEx(body);
    } else {
      Log.w(
          TAG,
          "Ignoring strf box for unsupported track type: " + Util.getTrackTypeString(trackType));
      return null;
    }
  }

  public final Format format;

  public StreamFormatChunk(Format format) {
    this.format = format;
  }

  @Override
  public int getType() {
    return AviExtractor.FOURCC_strf;
  }

  private static @Nullable AviChunk parseBitmapInfoHeader(ParsableByteArray body) {
    body.skipBytes(4); // biSize.
    int width = body.readLittleEndianInt();
    int height = body.readLittleEndianInt();
    body.skipBytes(4); // biPlanes (2 bytes), biBitCount (2 bytes).
    int compression = body.readLittleEndianInt();
    String mimeType = getMimeTypeFromCompression(compression);
    if (mimeType == null) {
      Log.w(TAG, "Ignoring track with unsupported compression " + compression);
      return null;
    }
    Format.Builder formatBuilder = new Format.Builder();
    formatBuilder.setWidth(width).setHeight(height).setSampleMimeType(mimeType);
    return new StreamFormatChunk(formatBuilder.build());
  }

  // Syntax defined by the WAVEFORMATEX structure. See
  // https://docs.microsoft.com/en-us/previous-versions/dd757713(v=vs.85).
  private static @Nullable AviChunk parseWaveFormatEx(ParsableByteArray body) {
    int formatTag = body.readLittleEndianUnsignedShort();
    @Nullable String mimeType = getMimeTypeFromTag(formatTag);
    if (mimeType == null) {
      Log.w(TAG, "Ignoring track with unsupported format tag " + formatTag);
      return null;
    }
    int channelCount = body.readLittleEndianUnsignedShort();
    int samplesPerSecond = body.readLittleEndianInt();
    body.skipBytes(6); // averageBytesPerSecond (4 bytes), nBlockAlign (2 bytes).
    int bitsPerSample = body.readLittleEndianUnsignedShort();
    int pcmEncoding = Util.getPcmEncoding(bitsPerSample);
    int cbSize = body.bytesLeft() > 0 ? body.readLittleEndianUnsignedShort() : 0;
    Format.Builder formatBuilder = new Format.Builder();
    formatBuilder
        .setSampleMimeType(mimeType)
        .setChannelCount(channelCount)
        .setSampleRate(samplesPerSecond);
    if (mimeType.equals(MimeTypes.AUDIO_RAW) && pcmEncoding != C.ENCODING_INVALID) {
      formatBuilder.setPcmEncoding(pcmEncoding);
    }
    if (mimeType.equals(MimeTypes.AUDIO_AAC) && cbSize > 0) {
      byte[] codecData = new byte[cbSize];
      body.readBytes(codecData, /* offset= */ 0, codecData.length);
      formatBuilder.setInitializationData(ImmutableList.of(codecData));
    }
    return new StreamFormatChunk(formatBuilder.build());
  }

  private static @Nullable String getMimeTypeFromTag(int tag) {
    switch (tag) {
      case 0x1: // WAVE_FORMAT_PCM
        return MimeTypes.AUDIO_RAW;
      case 0x55: // WAVE_FORMAT_MPEGLAYER3
        return MimeTypes.AUDIO_MPEG;
      case 0xff: // WAVE_FORMAT_AAC
        return MimeTypes.AUDIO_AAC;
      case 0x2000: // WAVE_FORMAT_DVM - AC3
        return MimeTypes.AUDIO_AC3;
      case 0x2001: // WAVE_FORMAT_DTS2
        return MimeTypes.AUDIO_DTS;
      default:
        return null;
    }
  }

  private static @Nullable String getMimeTypeFromCompression(int compression) {
    switch (compression) {
      case 0x3234504d: // MP42
        return MimeTypes.VIDEO_MP42;
      case 0x3334504d: // MP43
        return MimeTypes.VIDEO_MP43;
      case 0x34363248: // H264
      case 0x34363268: // h264
      case 0x31435641: // AVC1
      case 0x31637661: // avc1
      case 0x34363258: // X264
      case 0x34363278: // x264
        return MimeTypes.VIDEO_H264;
      case 0x58564944: // DIVX
      case 0x78766964: // divx
      case 0x30355844: // DX50
      case 0x30357864: // dx50
      case 0x44495658: // XVID
      case 0x64697678: // xvid
      case 0x5634504d: // MP4V
      case 0x7634706d: // mp4v
      case 0x34504d46: // FMP4
      case 0x34706d66: // fmp4
      case 0x44495633: // 3VID
        return MimeTypes.VIDEO_MP4V;
      case 0x47504a4d: // MJPG
      case 0x67706a6d: // mjpg
        return MimeTypes.VIDEO_MJPEG;
      default:
        return null;
    }
  }
}

package com.replayer.updater

import android.content.Context
import android.content.Intent
import android.util.Log
import androidx.core.content.FileProvider
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.OkHttpClient
import okhttp3.Request
import org.json.JSONArray
import org.json.JSONObject
import java.io.File
import java.io.FileOutputStream
import java.util.concurrent.TimeUnit

data class ReleaseInfo(
    val tagName: String,
    val versionName: String,
    val title: String,
    val changelog: String,
    val downloadUrl: String,
    val fileName: String,
    val fileSize: Long
)

sealed class UpdateCheckResult {
    data class UpdateAvailable(val release: ReleaseInfo) : UpdateCheckResult()
    object UpToDate : UpdateCheckResult()
    object NoReleasesFound : UpdateCheckResult()
    data class Error(val message: String) : UpdateCheckResult()
}

class UpdateManager(private val context: Context) {
    private val client = OkHttpClient.Builder()
        .connectTimeout(15, TimeUnit.SECONDS)
        .readTimeout(60, TimeUnit.SECONDS)
        .followRedirects(true)
        .followSslRedirects(true)
        .build()

    private val repoOwner = "SuperiorMor"
    private val repoName = "Re-Player"

    suspend fun checkForUpdates(currentVersion: String): UpdateCheckResult = withContext(Dispatchers.IO) {
        try {
            val url = "https://api.github.com/repos/$repoOwner/$repoName/releases/latest"
            val request = Request.Builder()
                .url(url)
                .header("Accept", "application/vnd.github.v3+json")
                .header("User-Agent", "RePlayer-App/v$currentVersion (Android TV)")
                .build()

            val response = client.newCall(request).execute()
            if (response.code == 404) {
                return@withContext UpdateCheckResult.NoReleasesFound
            }
            if (response.code == 403 || !response.isSuccessful) {
                // Rate limited (403) or API error -> Fallback to web redirect
                Log.w("UpdateManager", "GitHub API returned ${response.code}, falling back to web release check")
                return@withContext fetchLatestReleaseViaHtmlFallback(currentVersion)
            }

            val body = response.body?.string()
            if (body.isNullOrBlank()) {
                return@withContext fetchLatestReleaseViaHtmlFallback(currentVersion)
            }

            val json = JSONObject(body)
            val tagName = json.optString("tag_name", "")
            val cleanVersion = tagName.removePrefix("v").trim()
            val name = json.optString("name", "Re:Player $tagName")
            val changelog = json.optString("body", "Список изменений отсутствует")

            val assets = json.optJSONArray("assets") ?: JSONArray()
            var apkUrl: String? = null
            var apkName = "RePlayer-update.apk"
            var apkSize = 0L

            for (i in 0 until assets.length()) {
                val asset = assets.getJSONObject(i)
                val assetName = asset.optString("name", "")
                if (assetName.endsWith(".apk", ignoreCase = true)) {
                    apkUrl = asset.optString("browser_download_url")
                    apkName = assetName
                    apkSize = asset.optLong("size", 0L)
                    break
                }
            }

            if (apkUrl.isNullOrEmpty()) {
                apkName = "RePlayer-$tagName.apk"
                apkUrl = "https://github.com/$repoOwner/$repoName/releases/download/$tagName/$apkName"
            }

            val releaseInfo = ReleaseInfo(
                tagName = tagName,
                versionName = cleanVersion,
                title = name,
                changelog = changelog,
                downloadUrl = apkUrl,
                fileName = apkName,
                fileSize = apkSize
            )

            if (isNewerVersion(cleanVersion, currentVersion)) {
                UpdateCheckResult.UpdateAvailable(releaseInfo)
            } else {
                UpdateCheckResult.UpToDate
            }
        } catch (e: Exception) {
            Log.e("UpdateManager", "Check update via API failed, trying web fallback", e)
            fetchLatestReleaseViaHtmlFallback(currentVersion)
        }
    }

    private fun fetchLatestReleaseViaHtmlFallback(currentVersion: String): UpdateCheckResult {
        return try {
            val redirectClient = client.newBuilder()
                .followRedirects(false)
                .build()

            val webUrl = "https://github.com/$repoOwner/$repoName/releases/latest"
            val webRequest = Request.Builder()
                .url(webUrl)
                .header("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) RePlayer-TV-App/$currentVersion")
                .head()
                .build()

            val webResponse = redirectClient.newCall(webRequest).execute()
            var targetTag: String? = null
            val location = webResponse.header("Location")
            if (!location.isNullOrBlank() && location.contains("/releases/tag/")) {
                targetTag = location.substringAfterLast("/")
            }

            if (targetTag.isNullOrBlank()) {
                val fullResponse = client.newCall(
                    Request.Builder()
                        .url(webUrl)
                        .header("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) RePlayer-TV-App/$currentVersion")
                        .build()
                ).execute()
                val finalUrl = fullResponse.request.url.toString()
                if (finalUrl.contains("/releases/tag/")) {
                    targetTag = finalUrl.substringAfterLast("/")
                }
            }

            if (targetTag.isNullOrBlank()) {
                return UpdateCheckResult.Error("Не удалось определить последнюю версию с GitHub")
            }

            val cleanVersion = targetTag.removePrefix("v").trim()
            val apkName = "RePlayer-v$cleanVersion.apk"
            val apkUrl = "https://github.com/$repoOwner/$repoName/releases/download/$targetTag/$apkName"

            val releaseInfo = ReleaseInfo(
                tagName = targetTag,
                versionName = cleanVersion,
                title = "Re:Player $targetTag",
                changelog = "Обновление доступно для установки",
                downloadUrl = apkUrl,
                fileName = apkName,
                fileSize = 0L
            )

            if (isNewerVersion(cleanVersion, currentVersion)) {
                UpdateCheckResult.UpdateAvailable(releaseInfo)
            } else {
                UpdateCheckResult.UpToDate
            }
        } catch (e: Exception) {
            Log.e("UpdateManager", "Fallback release check failed", e)
            UpdateCheckResult.Error(e.localizedMessage ?: "Ошибка сети при проверке обновлений")
        }
    }

    suspend fun downloadApk(
        downloadUrl: String,
        fileName: String = "RePlayer-update.apk",
        onProgress: (Float) -> Unit
    ): File? = withContext(Dispatchers.IO) {
        try {
            val dir = context.externalCacheDir ?: context.cacheDir
            val updatesDir = File(dir, "updates").apply { mkdirs() }
            val targetFile = File(updatesDir, fileName)
            if (targetFile.exists()) targetFile.delete()

            val request = Request.Builder()
                .url(downloadUrl)
                .header("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) RePlayer-TV-App")
                .build()
            val response = client.newCall(request).execute()
            if (!response.isSuccessful) return@withContext null

            val body = response.body ?: return@withContext null
            val totalBytes = body.contentLength()

            body.byteStream().use { input ->
                FileOutputStream(targetFile).use { output ->
                    val buffer = ByteArray(8192)
                    var bytesRead: Int
                    var totalRead = 0L

                    while (input.read(buffer).also { bytesRead = it } != -1) {
                        output.write(buffer, 0, bytesRead)
                        totalRead += bytesRead
                        if (totalBytes > 0) {
                            onProgress((totalRead.toFloat() / totalBytes.toFloat()).coerceIn(0f, 1f))
                        }
                    }
                    output.flush()
                }
            }

            onProgress(1.0f)
            targetFile
        } catch (e: Exception) {
            Log.e("UpdateManager", "Download failed", e)
            null
        }
    }

    fun installApk(apkFile: File) {
        try {
            val uri = FileProvider.getUriForFile(
                context,
                "${context.packageName}.fileprovider",
                apkFile
            )

            val intent = Intent(Intent.ACTION_VIEW).apply {
                setDataAndType(uri, "application/vnd.android.package-archive")
                addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }

            context.startActivity(intent)
        } catch (e: Exception) {
            Log.e("UpdateManager", "Install intent failed", e)
        }
    }

    fun isNewerVersion(remote: String, current: String): Boolean {
        val remoteParts = remote.split('.').mapNotNull { it.toIntOrNull() }
        val currentParts = current.split('.').mapNotNull { it.toIntOrNull() }

        val maxLen = maxOf(remoteParts.size, currentParts.size)
        for (i in 0 until maxLen) {
            val r = remoteParts.getOrElse(i) { 0 }
            val c = currentParts.getOrElse(i) { 0 }
            if (r > c) return true
            if (r < c) return false
        }
        return false
    }
}

@OptIn(UnstableApi::class)
@Composable
fun PlayerScreen(
    mediaItem: MediaItem,
    onBack: () -> Unit,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val engine = remember { RePlayerEngine(context) }
    val playerState by engine.playerState.collectAsState()

    var showControls by remember { mutableStateOf(true) }
    var showAudioDialog by remember { mutableStateOf(false) }
    var showSubtitleDialog by remember { mutableStateOf(false) }
    var lastInteractionTime by remember { mutableLongStateOf(System.currentTimeMillis()) }

    val playButtonFocusRequester = remember { FocusRequester() }
    val rootFocusRequester = remember { FocusRequester() }

    // KEEP SCREEN ON during video playback!
    DisposableEffect(Unit) {
        val activity = context as? Activity
        activity?.window?.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)

        engine.play(mediaItem)

        onDispose {
            activity?.window?.clearFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
            engine.release()
        }
    }

    // Auto-hide TV controls after 5 seconds of inactivity
    LaunchedEffect(showControls, lastInteractionTime) {
        if (showControls) {
            delay(5000)
            showControls = false
        }
    }

    // Focus management: when controls are shown, focus play button; when hidden, focus root box
    LaunchedEffect(showControls) {
        if (showControls) {
            delay(100)
            try {
                playButtonFocusRequester.requestFocus()
            } catch (_: Exception) {
                // Ignore focus request failure
            }
        } else {
            delay(50)
            try {
                rootFocusRequester.requestFocus()
            } catch (_: Exception) {
                // Ignore focus request failure
            }
        }
    }

    BackHandler(enabled = true) {
        if (showAudioDialog || showSubtitleDialog) {
            showAudioDialog = false
            showSubtitleDialog = false
            lastInteractionTime = System.currentTimeMillis()
        } else if (showControls) {
            showControls = false
            try {
                rootFocusRequester.requestFocus()
            } catch (_: Exception) {}
        } else {
            onBack()
        }
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(Color.Black)
            .focusRequester(rootFocusRequester)
            .focusable()
            .onPreviewKeyEvent { keyEvent ->
                if (keyEvent.type == KeyEventType.KeyDown && (keyEvent.key == Key.Back || keyEvent.key == Key.Escape)) {
                    if (showAudioDialog || showSubtitleDialog) {
                        showAudioDialog = false
                        showSubtitleDialog = false
                        lastInteractionTime = System.currentTimeMillis()
                        return@onPreviewKeyEvent true
                    } else if (showControls) {
                        showControls = false
                        try {
                            rootFocusRequester.requestFocus()
                        } catch (_: Exception) {}
                        return@onPreviewKeyEvent true
                    } else {
                        onBack()
                        return@onPreviewKeyEvent true
                    }
                }
                false
            }
            .onKeyEvent { keyEvent ->
                if (keyEvent.type == KeyEventType.KeyDown) {
                    if (!showControls) {
                        when (keyEvent.key) {
                            Key.DirectionRight -> {
                                // 10s forward without showing controls
                                engine.seekRelative(10000)
                                true
                            }
                            Key.DirectionLeft -> {
                                // 10s back without showing controls
                                engine.seekRelative(-10000)
                                true
                            }
                            Key.DirectionCenter, Key.Enter,
                            Key.DirectionUp, Key.DirectionDown,
                            Key.MediaPlay, Key.MediaPause, Key.MediaPlayPause,
                            Key.Spacebar -> {
                                showControls = true
                                lastInteractionTime = System.currentTimeMillis()
                                true
                            }
                            else -> false
                        }
                    } else {
                        lastInteractionTime = System.currentTimeMillis()
                        false
                    }
                } else false
            }
    ) {
        // Video Surface
        AndroidView(
            factory = { ctx ->
                PlayerView(ctx).apply {
                    player = engine.exoPlayer
                    useController = false
                    keepScreenOn = true
                    isFocusable = false
                    isFocusableInTouchMode = false
                }
            },
            modifier = Modifier
                .fillMaxSize()
                .clickable(
                    interactionSource = remember { MutableInteractionSource() },
                    indication = null
                ) {
                    showControls = !showControls
                    lastInteractionTime = System.currentTimeMillis()
                }
        )

        // TV On-Screen Display (OSD) Overlay
        AnimatedVisibility(
            visible = showControls && !showAudioDialog && !showSubtitleDialog,
            enter = fadeIn(),
            exit = fadeOut()
        ) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color(0xB3000000))
                    .padding(horizontal = 48.dp, vertical = 36.dp)
            ) {
                // Top Bar: Title & Source & Active Audio Track
                Column(modifier = Modifier.align(Alignment.TopStart)) {
                    Text(
                        text = playerState.currentMediaTitle.ifEmpty { mediaItem.title },
                        style = MaterialTheme.typography.titleLarge,
                        color = TextPrimary
                    )
                    Spacer(modifier = Modifier.height(4.dp))
                    val selectedAudio = playerState.audioTracks.find { it.id == playerState.selectedAudioTrackId }
                    val audioInfo = selectedAudio?.let { " • Звук: ${it.label}" } ?: ""
                    Text(
                        text = "Источник: ${mediaItem.sourceType.name}$audioInfo",
                        style = MaterialTheme.typography.bodyMedium,
                        color = AccentCyan
                    )
                }

                // Bottom Controls: Progress & Audio/Subtitle selection buttons
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .align(Alignment.BottomCenter)
                ) {
                    // Time Labels & Interactive Seek Bar
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(
                            text = formatTime(playerState.currentPositionMs),
                            style = MaterialTheme.typography.bodyMedium,
                            color = TextPrimary
                        )

                        Spacer(modifier = Modifier.width(16.dp))

                        // Interactive Focusable Timeline Seek Bar
                        TvTimelineBar(
                            currentPositionMs = playerState.currentPositionMs,
                            durationMs = playerState.durationMs,
                            onSeekRelative = { delta ->
                                engine.seekRelative(delta)
                                lastInteractionTime = System.currentTimeMillis()
                            },
                            modifier = Modifier.weight(1f)
                        )

                        Spacer(modifier = Modifier.width(16.dp))

                        Text(
                            text = formatTime(playerState.durationMs),
                            style = MaterialTheme.typography.bodyMedium,
                            color = TextSecondary
                        )
                    }

                    Spacer(modifier = Modifier.height(20.dp))

                    // Buttons Row for TV Remote
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        TvPlayerIconButton(
                            icon = Icons.Default.Replay10,
                            contentDescription = "Назад на 10 сек",
                            onClick = {
                                engine.seekRelative(-10000)
                                lastInteractionTime = System.currentTimeMillis()
                            }
                        )

                        Spacer(modifier = Modifier.width(20.dp))

                        TvPlayerIconButton(
                            icon = if (playerState.isPlaying) Icons.Default.Pause else Icons.Default.PlayArrow,
                            contentDescription = if (playerState.isPlaying) "Пауза" else "Воспроизведение",
                            focusRequester = playButtonFocusRequester,
                            onClick = {
                                engine.togglePlayPause()
                                lastInteractionTime = System.currentTimeMillis()
                            }
                        )

                        Spacer(modifier = Modifier.width(20.dp))

                        TvPlayerIconButton(
                            icon = Icons.Default.Forward10,
                            contentDescription = "Вперед на 10 сек",
                            onClick = {
                                engine.seekRelative(10000)
                                lastInteractionTime = System.currentTimeMillis()
                            }
                        )

                        Spacer(modifier = Modifier.width(36.dp))

                        // Audio Track Selector Button
                        TvPlayerIconButton(
                            icon = Icons.Default.Audiotrack,
                            contentDescription = "Аудиодорожки",
                            onClick = {
                                showAudioDialog = true
                                lastInteractionTime = System.currentTimeMillis()
                            }
                        )

                        Spacer(modifier = Modifier.width(16.dp))

                        // Subtitle Track Selector Button
                        TvPlayerIconButton(
                            icon = Icons.Default.ClosedCaption,
                            contentDescription = "Субтитры",
                            onClick = {
                                showSubtitleDialog = true
                                lastInteractionTime = System.currentTimeMillis()
                            }
                        )
                    }
                }
            }
        }

        // Audio Track Selection Dialog
        if (showAudioDialog) {
            AudioTrackDialog(
                tracks = playerState.audioTracks,
                selectedId = playerState.selectedAudioTrackId,
                onSelect = { track ->
                    engine.selectAudioTrack(track)
                    showAudioDialog = false
                    lastInteractionTime = System.currentTimeMillis()
                },
                onDismiss = { showAudioDialog = false }
            )
        }

        // Subtitle Track Selection Dialog
        if (showSubtitleDialog) {
            SubtitleTrackDialog(
                tracks = playerState.subtitleTracks,
                selectedId = playerState.selectedSubtitleTrackId,
                onSelect = { track ->
                    engine.selectSubtitleTrack(track)
                    showSubtitleDialog = false
                    lastInteractionTime = System.currentTimeMillis()
                },
                onDismiss = { showSubtitleDialog = false }
            )
        }
    }
}

@Composable
fun TvTimelineBar(
    currentPositionMs: Long,
    durationMs: Long,
    onSeekRelative: (Long) -> Unit,
    modifier: Modifier = Modifier
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()
    val progress = if (durationMs > 0) (currentPositionMs.toFloat() / durationMs.toFloat()).coerceIn(0f, 1f) else 0f

    BoxWithConstraints(
        modifier = modifier
            .height(36.dp)
            .focusable(interactionSource = interactionSource)
            .onKeyEvent { keyEvent ->
                if (keyEvent.type == KeyEventType.KeyDown) {
                    when (keyEvent.key) {
                        Key.DirectionLeft -> {
                            onSeekRelative(-10000)
                            true
                        }
                        Key.DirectionRight -> {
                            onSeekRelative(10000)
                            true
                        }
                        else -> false
                    }
                } else false
            },
        contentAlignment = Alignment.Center
    ) {
        val totalWidth = maxWidth

        // Background track bar
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(if (isFocused) 10.dp else 6.dp)
                .clip(RoundedCornerShape(4.dp))
                .background(if (isFocused) Color(0x66FFFFFF) else Color(0x33FFFFFF))
        ) {
            // Played progress fill
            Box(
                modifier = Modifier
                    .fillMaxWidth(fraction = progress)
                    .fillMaxHeight()
                    .background(AccentCyan)
            )
        }

        // Focused Scrubber Thumb
        if (isFocused) {
            val thumbOffset = (totalWidth - 16.dp) * progress
            Box(
                modifier = Modifier.fillMaxWidth(),
                contentAlignment = Alignment.CenterStart
            ) {
                Box(
                    modifier = Modifier
                        .padding(start = thumbOffset.coerceAtLeast(0.dp))
                        .size(16.dp)
                        .clip(RoundedCornerShape(8.dp))
                        .background(Color.White)
                        .border(2.dp, AccentCyan, RoundedCornerShape(8.dp))
                )
            }
        }
    }
}

@Composable
fun TvPlayerIconButton(
    icon: ImageVector,
    contentDescription: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    focusRequester: FocusRequester? = null
) {
    val interactionSource = remember { MutableInteractionSource() }
    val isFocused by interactionSource.collectIsFocusedAsState()
    val scale by animateFloatAsState(if (isFocused) 1.15f else 1.0f, label = "btn_scale")

    Box(
        modifier = modifier
            .size(56.dp)
            .padding(4.dp)
            .scale(scale)
            .clip(RoundedCornerShape(12.dp))
            .background(if (isFocused) AccentCyan.copy(alpha = 0.25f) else Color(0x33FFFFFF))
            .then(
                if (isFocused) {
                    Modifier.border(2.5.dp, FocusBorder, RoundedCornerShape(12.dp))
                } else {
                    Modifier.border(1.dp, Color(0x22FFFFFF), RoundedCornerShape(12.dp))
                }
            )
            .then(if (focusRequester != null) Modifier.focusRequester(focusRequester) else Modifier)
            .clickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = onClick
            )
            .focusable(interactionSource = interactionSource),
        contentAlignment = Alignment.Center
    ) {
        Icon(
            imageVector = icon,

            tint = if (isFocused) AccentCyan else TextPrimary,

private fun formatTime(ms: Long): String {
    val totalSeconds = (ms / 1000).coerceAtLeast(0)
    val seconds = totalSeconds % 60
    val minutes = (totalSeconds / 60) % 60
    val hours = totalSeconds / 3600
    return if (hours > 0) {
        String.format("%02d:%02d:%02d", hours, minutes, seconds)
    } else {
        String.format("%02d:%02d", minutes, seconds)
    }
}

    val context = LocalContext.current

    var isLoadingFolder by remember { mutableStateOf(false) }

    // User preferences: Default view is LIST

    var sortMode by remember { mutableStateOf(SortMode.BY_NAME) }

    val listState = rememberLazyListState()
    val gridState = rememberLazyGridState()

    val firstItemFocusRequester = remember { FocusRequester() }
    val targetItemFocusRequester = remember { FocusRequester() }
    var lastExitTimestamp by remember { mutableLongStateOf(0L) }

    fun loadFolder(controlUrl: String, folderId: String) {
        isLoadingFolder = true
        scope.launch {
            currentItems = contentDirectory.browse(controlUrl, folderId)
            isLoadingFolder = false
        }
    }

    fun openFolder(controlUrl: String, folderId: String, title: String) {
        onConsumeLastPlayedMediaId()
        folderStack = folderStack + Pair(folderId, title)
        loadFolder(controlUrl, folderId)

            val controlUrl = connectedDevice!!.contentDirectoryControlUrl ?: return@LaunchedEffect
            if (initialFolder != null) {
                if (folderStack.lastOrNull()?.first != initialFolder.first) {
                    folderStack = listOf(Pair("0", connectedDevice!!.friendlyName), initialFolder)
                    loadFolder(controlUrl, initialFolder.first)
                    onFolderLoaded()
                }
            } else if (folderStack.isEmpty()) {
                folderStack = listOf(Pair("0", connectedDevice!!.friendlyName))
                loadFolder(controlUrl, "0")
            }
        }
    }

            val parentIdx = folderStack.size - 2

            folderStack = folderStack.take(parentIdx + 1)

                loadFolder(url, parentFolder.first)

            onBackToHome()

                                            .background(AccentCyan.copy(alpha = 0.2f))
                                            .padding(horizontal = 6.dp, vertical = 2.dp)
                                    ) {
                                        Text(
                                            text = item.fileFormat(),
                                            color = AccentCyan,
                                            fontSize = 11.sp,
                                            fontWeight = FontWeight.Bold
                                        )
                                    }
                                }
                            }
                            Spacer(modifier = Modifier.height(12.dp))
                            Text(
                                text = item.title,
                                style = MaterialTheme.typography.titleMedium,
                                maxLines = 1,
                                overflow = TextOverflow.Ellipsis
                            )
                            val sizeStr = item.formattedSize()
                            if (sizeStr.isNotEmpty()) {
                                Text(
                                    text = sizeStr,
                                    style = MaterialTheme.typography.bodySmall,
                                    color = TextSecondary
                                )
                            }
                        }
                    }
                }
                }
            }

            if (isLoadingFolder) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(Color.Black.copy(alpha = 0.45f)),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator(color = AccentCyan)
                }
            }
        }
    }

    // 1. Returning from player: focus the played video item and keep focus on it!
    LaunchedEffect(isPlayerActive) {
        if (!isPlayerActive && lastPlayedMediaId != null) {
            val targetIndex = sortedItems.indexOfFirst { it.id == lastPlayedMediaId }
            if (targetIndex >= 0) {
                delay(60)
                try {
                    if (viewMode == ViewMode.LIST) {
                        listState.scrollToItem(maxOf(0, targetIndex))
                    } else {
                        gridState.scrollToItem(maxOf(0, targetIndex))
                    }
                } catch (_: Exception) {}
                delay(80)
                try {
                    targetItemFocusRequester.requestFocus()
                } catch (_: Exception) {}
            }
        }
    }

    // 2. Folder navigation (entering folder or pressing Back): focus first item
    LaunchedEffect(folderStack) {
        if (folderStack.isNotEmpty() && !isPlayerActive) {
            delay(60)
            try {
                if (viewMode == ViewMode.LIST) {
                    listState.scrollToItem(0)
                } else {
                    gridState.scrollToItem(0)
                }
                firstItemFocusRequester.requestFocus()
            } catch (_: Exception) {}
        }
    }

    // 3. Initial folder items loaded
    LaunchedEffect(currentItems) {
        if (!isPlayerActive && lastPlayedMediaId == null && currentItems.isNotEmpty()) {
            delay(50)
            try {
                firstItemFocusRequester.requestFocus()
            } catch (_: Exception) {}
        }
    }
}

@Composable
fun GoogleDriveScreen(
    onPlayMedia: (MediaItem) -> Unit,
    onNavigateToSettings: () -> Unit,
    onBackToHome: () -> Unit,
    modifier: Modifier = Modifier,
    initialFolder: Pair<String, String>? = null,
    lastPlayedMediaId: String? = null,
    isPlayerActive: Boolean = false,
    onFolderLoaded: () -> Unit = {},
    onConsumeLastPlayedMediaId: () -> Unit = {}
) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    val authManager = remember { GoogleDriveAuthManager(context) }
    val driveService = remember { GoogleDriveService(authManager) }
    val favoritesManager = remember { FavoritesManager(context) }

    val isAuthenticated by GoogleDriveAuthManager.sharedAuthenticatedFlow.collectAsState()
    val favorites by favoritesManager.favorites.collectAsState()

    var currentItems by remember { mutableStateOf<List<MediaItem>>(emptyList()) }
    var folderStack by remember { mutableStateOf<List<Pair<String, String>>>(emptyList()) }
    var isLoading by remember { mutableStateOf(false) }
    var driveLoadError by remember { mutableStateOf<String?>(null) }

    var viewMode by remember { mutableStateOf(ViewMode.LIST) }
    var sortMode by remember { mutableStateOf(SortMode.BY_NAME) }

    val listState = rememberLazyListState()
    val gridState = rememberLazyGridState()

    val firstItemFocusRequester = remember { FocusRequester() }
    val targetItemFocusRequester = remember { FocusRequester() }
    var lastExitTimestamp by remember { mutableLongStateOf(0L) }

    fun loadFolder(folderId: String) {
        isLoading = true
        driveLoadError = null
        scope.launch {
            currentItems = driveService.listFolder(folderId)
            driveLoadError = driveService.lastErrorMessage
            isLoading = false
        }
    }

    fun openFolder(folderId: String, title: String) {
        onConsumeLastPlayedMediaId()
        folderStack = folderStack + Pair(folderId, title)
        loadFolder(folderId)
    }

    // Initialize or switch folder when isAuthenticated or initialFolder changes
    LaunchedEffect(isAuthenticated, initialFolder) {
        if (isAuthenticated) {
            if (initialFolder != null) {
                if (folderStack.lastOrNull()?.first != initialFolder.first) {
                    folderStack = listOf(Pair("root", "Мой диск"), initialFolder)
                    loadFolder(initialFolder.first)
                    onFolderLoaded()
                }
            } else if (folderStack.isEmpty()) {
                folderStack = listOf(Pair("root", "Мой диск"))
                loadFolder("root")
            }
        }
    }

    fun navigateBack() {
        onConsumeLastPlayedMediaId()
        if (folderStack.size > 1) {
            val parentIdx = folderStack.size - 2
            val parentFolder = folderStack[parentIdx]
            folderStack = folderStack.take(parentIdx + 1)
            loadFolder(parentFolder.first)
        } else {
            onBackToHome()
        }
    }

    BackHandler(enabled = !isPlayerActive) {
        // Prevent accidental multiple back taps within 400ms after closing player from popping folders
        if (System.currentTimeMillis() - lastExitTimestamp < 400) {
            return@BackHandler
        }
        navigateBack()
    }

    val sortedItems = remember(currentItems, sortMode) {
        when (sortMode) {
            SortMode.BY_NAME -> {
                currentItems.sortedWith(
                    compareBy<MediaItem> { it.type != MediaType.DIRECTORY }
                        .thenBy(String.CASE_INSENSITIVE_ORDER) { it.title }
                )
            }
            SortMode.BY_DATE -> {
                currentItems.sortedWith(
                    compareBy<MediaItem> { it.type != MediaType.DIRECTORY }
                        .thenByDescending { it.dateAdded ?: "" }
                        .thenBy(String.CASE_INSENSITIVE_ORDER) { it.title }
                )
            }
        }
    }

    // 1. Returning from player: focus the played video item and keep focus on it!
    LaunchedEffect(isPlayerActive) {
        if (!isPlayerActive && lastPlayedMediaId != null) {
            val targetIndex = sortedItems.indexOfFirst { it.id == lastPlayedMediaId }

                try {
                    if (viewMode == ViewMode.LIST) {
                        listState.scrollToItem(maxOf(0, targetIndex))
                    } else {
                        gridState.scrollToItem(maxOf(0, targetIndex))
                    }
                } catch (_: Exception) {}
                delay(80)
                try {
                    targetItemFocusRequester.requestFocus()
                } catch (_: Exception) {}
            }
        }
    }

    // 2. Folder navigation (entering folder or pressing Back): focus first item
    LaunchedEffect(folderStack) {
        if (folderStack.isNotEmpty() && !isPlayerActive) {
            delay(60)
            try {
                if (viewMode == ViewMode.LIST) {
                    listState.scrollToItem(0)
                } else {
                    gridState.scrollToItem(0)
                }
                firstItemFocusRequester.requestFocus()
            } catch (_: Exception) {}
        }
    }

    // 3. Initial folder items loaded
    LaunchedEffect(currentItems) {
        if (!isPlayerActive && lastPlayedMediaId == null && currentItems.isNotEmpty()) {
            delay(50)
            try {
                firstItemFocusRequester.requestFocus()
            } catch (_: Exception) {}
        }
    }

    if (!isAuthenticated) {
        // Not connected state
        Box(
            modifier = modifier
                .fillMaxSize()
                .padding(32.dp),
            contentAlignment = Alignment.Center
        ) {
            Column(
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.Center
            ) {
                Box(
                    modifier = Modifier
                        .size(80.dp)
                        .clip(RoundedCornerShape(20.dp))
                        .background(DarkCard),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = Icons.Default.Cloud,
                        contentDescription = null,
                        tint = Color(0xFF4285F4),
                        modifier = Modifier.size(42.dp)
                    )
                }
                Spacer(modifier = Modifier.height(20.dp))
                Text(
                    text = "Подключите свой Google Drive в настройках",
                    style = MaterialTheme.typography.titleLarge.copy(fontWeight = FontWeight.SemiBold),
                    color = TextPrimary
                )
                Spacer(modifier = Modifier.height(8.dp))
                Text(
                    text = "Для безопасного доступа к файлам облака подключите аккаунт в меню настроек.",
                    style = MaterialTheme.typography.bodyMedium,
                    color = TextSecondary
                )
                Spacer(modifier = Modifier.height(24.dp))
                TvFocusCard(
                    onClick = onNavigateToSettings,
                    modifier = Modifier
                        .width(220.dp)
                        .height(48.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxSize(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Icon(Icons.Default.Settings, contentDescription = null, tint = AccentCyan, modifier = Modifier.size(20.dp))
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(text = "Перейти в настройки", color = AccentCyan, style = MaterialTheme.typography.bodyMedium)
                    }
                }
            }
        }
        return
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(horizontal = 32.dp, vertical = 24.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Column(modifier = Modifier.weight(1f)) {
                Text(
                    text = "Google Drive",
                    style = MaterialTheme.typography.displayLarge
                )
                val breadcrumb = folderStack.joinToString(" / ") { it.second }
                Text(
                    text = breadcrumb.ifEmpty { "Мой диск" },
                    style = MaterialTheme.typography.bodyLarge,
                    color = AccentCyan,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
            }

            Row(
                horizontalArrangement = Arrangement.spacedBy(12.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                // View Switcher (List / Grid)
                TvFocusCard(
                    onClick = {
                        viewMode = if (viewMode == ViewMode.LIST) ViewMode.GRID else ViewMode.LIST
                    },
                    modifier = Modifier
                        .width(160.dp)
                        .height(48.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxSize(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Icon(
                            imageVector = if (viewMode == ViewMode.LIST) Icons.Default.ViewList else Icons.Default.GridView,
                            contentDescription = null,
                            tint = AccentCyan,
                            modifier = Modifier.size(20.dp)
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(
                            text = if (viewMode == ViewMode.LIST) "Вид: Список" else "Вид: Плитка",
                            style = MaterialTheme.typography.bodyMedium,
                            color = TextPrimary
                        )
                    }
                }

                // Sort Switcher (Name / Date)
                TvFocusCard(
                    onClick = {
                        sortMode = if (sortMode == SortMode.BY_NAME) SortMode.BY_DATE else SortMode.BY_NAME
                    },
                    modifier = Modifier
                        .width(170.dp)
                        .height(48.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxSize(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Icon(
                            imageVector = Icons.Default.Sort,
                            contentDescription = null,
                            tint = AccentCyan,
                            modifier = Modifier.size(20.dp)
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        Text(
                            text = if (sortMode == SortMode.BY_NAME) "Сорт: По имени" else "Сорт: По дате",
                            style = MaterialTheme.typography.bodyMedium,
                            color = TextPrimary
                        )
                    }
                }
            }
        }

        Spacer(modifier = Modifier.height(20.dp))

        Box(modifier = Modifier.fillMaxSize()) {
            if (sortedItems.isEmpty() && !isLoading) {
                Box(
                    modifier = Modifier.fillMaxSize(),
                    contentAlignment = Alignment.Center
                ) {
                    val currentFolderId = folderStack.lastOrNull()?.first ?: "root"
                    if (driveLoadError != null) {
                        Column(
                            horizontalAlignment = Alignment.CenterHorizontally,
                            verticalArrangement = Arrangement.Center,
                            modifier = Modifier.padding(32.dp)
                        ) {
                            Text(
                                text = driveLoadError ?: "Ошибка загрузки",
                                style = MaterialTheme.typography.titleMedium,
                                color = Color(0xFFFF5252)
                            )
                            Spacer(modifier = Modifier.height(16.dp))
                            TvFocusCard(
                                onClick = { loadFolder(currentFolderId) },
                                modifier = Modifier
                                    .width(180.dp)
                                    .height(48.dp)
                                    .focusRequester(firstItemFocusRequester)
                            ) {
                                Row(
                                    modifier = Modifier.fillMaxSize(),
                                    horizontalArrangement = Arrangement.Center,
                                    verticalAlignment = Alignment.CenterVertically
                                ) {
                                    Icon(Icons.Default.Refresh, contentDescription = null, tint = AccentCyan, modifier = Modifier.size(20.dp))
                                    Spacer(modifier = Modifier.width(8.dp))
                                    Text(text = "Повторить", color = AccentCyan, style = MaterialTheme.typography.bodyMedium)
                                }
                            }
                        }
                    } else {
                        Column(
                            horizontalAlignment = Alignment.CenterHorizontally,
                            verticalArrangement = Arrangement.Center,
                            modifier = Modifier
                                .width(560.dp)
                                .clip(RoundedCornerShape(16.dp))
                                .background(DarkCard)
                                .padding(32.dp)
                        ) {
                            Box(
                                modifier = Modifier
                                    .size(64.dp)
                                    .clip(RoundedCornerShape(16.dp))
                                    .background(AccentCyan.copy(alpha = 0.15f)),
                                contentAlignment = Alignment.Center
                            ) {
                                Icon(
                                    imageVector = Icons.Default.Cloud,
                                    contentDescription = null,
                                    tint = AccentCyan,
                                    modifier = Modifier.size(36.dp)
                                )

                            Spacer(modifier = Modifier.height(16.dp))
                            Text(
                                text = if (currentFolderId == "root") "Google Диск подключен" else "Папка пуста",
                                style = MaterialTheme.typography.titleLarge.copy(fontWeight = FontWeight.Bold),
                                color = TextPrimary

                            Spacer(modifier = Modifier.height(8.dp))
                            Text(
                                text = if (currentFolderId == "root")
                                    "В корне вашего Google Диска пока нет видеофайлов или папок. Загрузите видео на диск и нажмите «Обновить»."
                                else
                                    "В этой папке нет видео- или аудиофайлов. Скопируйте медиафайлы или вернитесь назад.",
                                style = MaterialTheme.typography.bodyMedium,
                                color = TextSecondary,
                                textAlign = TextAlign.Center
                            )
                            Spacer(modifier = Modifier.height(20.dp))
                            TvFocusCard(
                                onClick = { loadFolder(currentFolderId) },
                                modifier = Modifier
                                    .width(200.dp)
                                    .height(48.dp)
                                    .focusRequester(firstItemFocusRequester)
                            ) {
                                Row(
                                    modifier = Modifier.fillMaxSize(),
                                    horizontalArrangement = Arrangement.Center,
                                    verticalAlignment = Alignment.CenterVertically
                                ) {
                                    Icon(Icons.Default.Refresh, contentDescription = null, tint = AccentCyan, modifier = Modifier.size(20.dp))
                                    Spacer(modifier = Modifier.width(8.dp))
                                    Text(text = "Обновить список", color = AccentCyan, style = MaterialTheme.typography.bodyMedium, fontWeight = FontWeight.SemiBold)
                                }
                            }
                        }
                    }
                }
            } else if (viewMode == ViewMode.LIST) {
                // Detailed List View
                LazyColumn(
                    state = listState,
                    verticalArrangement = Arrangement.spacedBy(10.dp),
                    contentPadding = PaddingValues(vertical = 8.dp),
                    modifier = Modifier.fillMaxSize()
                ) {
                items(sortedItems.size, key = { sortedItems[it].id }) { index ->
                    val item = sortedItems[index]
                    val isFirst = index == 0
                    val isTarget = item.id == lastPlayedMediaId
                    val isFav = favorites.any { it.id == item.id }

                    val itemFocusModifier = when {
                        isTarget -> Modifier.focusRequester(targetItemFocusRequester)
                        isFirst -> Modifier.focusRequester(firstItemFocusRequester)
                        else -> Modifier
                    }

                    TvFocusCard(
                        onClick = {
                            if (item.type == MediaType.DIRECTORY) {
                                openFolder(item.id, item.title)
                            } else {
                                onPlayMedia(item)
                            }
                        },
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(78.dp)
                            .then(itemFocusModifier)
                    ) {
                        Row(
                            modifier = Modifier
                                .fillMaxSize()
                                .padding(horizontal = 20.dp),
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            val icon = when (item.type) {
                                MediaType.DIRECTORY -> Icons.Default.Folder
                                MediaType.VIDEO -> Icons.Default.VideoFile
                                MediaType.AUDIO -> Icons.Default.AudioFile
                                MediaType.IMAGE -> Icons.Default.Image
                                MediaType.UNKNOWN -> Icons.Default.VideoFile
                            }
                            Box(
                                modifier = Modifier
                                    .size(44.dp)
                                    .clip(RoundedCornerShape(8.dp))
                                    .background(AccentCyan.copy(alpha = 0.12f)),

                            ) {

                                    imageVector = icon,
                                    contentDescription = null,
                                    tint = AccentCyan,
                                    modifier = Modifier.size(24.dp)
                                )
                            }

                            Spacer(modifier = Modifier.width(16.dp))

                            Column(modifier = Modifier.weight(1f)) {
                                Text(
                                    text = item.title,
                                    style = MaterialTheme.typography.titleMedium,
                                    maxLines = 1,

                                    color = TextPrimary

                                Spacer(modifier = Modifier.height(4.dp))

                                                FavoriteFolder(
                                                    id = item.id,

                                    contentAlignment = Alignment.Center
                                ) {
                                    Icon(
                                        imageVector = if (isFav) Icons.Default.Favorite else Icons.Default.FavoriteBorder,
                                        contentDescription = "Избранное",
                                        tint = if (isFav) Color(0xFFFF4081) else TextSecondary,
                                        modifier = Modifier.size(20.dp)
                                    )
                                }
                            }
                        }
                    }
                }
            }
        } else {
            // Grid View
            LazyVerticalGrid(
                state = gridState,
                columns = GridCells.Adaptive(minSize = 220.dp),
                horizontalArrangement = Arrangement.spacedBy(16.dp),
                verticalArrangement = Arrangement.spacedBy(16.dp),
                contentPadding = PaddingValues(vertical = 8.dp)
            ) {
                items(sortedItems.size, key = { sortedItems[it].id }) { index ->
                    val item = sortedItems[index]
                    val isFirst = index == 0
                    val isTarget = item.id == lastPlayedMediaId
                    val isFav = favorites.any { it.id == item.id }

                    val itemFocusModifier = when {
                        isTarget -> Modifier.focusRequester(targetItemFocusRequester)
                        isFirst -> Modifier.focusRequester(firstItemFocusRequester)
                        else -> Modifier
                    }

                    TvFocusCard(
                        onClick = {
                            if (item.type == MediaType.DIRECTORY) {
                                openFolder(item.id, item.title)
                            } else {
                                onPlayMedia(item)
                            }
                        },
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(130.dp)
                            .then(itemFocusModifier)
                    ) {
                        Column(
                            modifier = Modifier
                                .fillMaxSize()
                                .padding(16.dp),
                            verticalArrangement = Arrangement.Center
                        ) {
                            val icon = when (item.type) {
                                MediaType.DIRECTORY -> Icons.Default.Folder
                                MediaType.VIDEO -> Icons.Default.VideoFile
                                MediaType.AUDIO -> Icons.Default.AudioFile
                                MediaType.IMAGE -> Icons.Default.Image
                                MediaType.UNKNOWN -> Icons.Default.VideoFile
                            }
                            Row(
                                modifier = Modifier.fillMaxWidth(),
                                horizontalArrangement = Arrangement.SpaceBetween,
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                Icon(icon, contentDescription = null, tint = AccentCyan)

                                Row(verticalAlignment = Alignment.CenterVertically) {
                                    if (item.type == MediaType.DIRECTORY) {
                                        Icon(
                                            imageVector = if (isFav) Icons.Default.Favorite else Icons.Default.FavoriteBorder,
                                            contentDescription = null,
                                            tint = if (isFav) Color(0xFFFF4081) else TextSecondary,
                                            modifier = Modifier
                                                .size(20.dp)
                                                .clickable {
                                                    favoritesManager.toggleFavorite(
                                                        FavoriteFolder(
                                                            id = item.id,
                                                            title = item.title,
                                                            sourceType = MediaSourceType.GOOGLE_DRIVE,
                                                            parentId = item.parentId
                                                        )
                                                    )
                                                }
                                        )
                                        Spacer(modifier = Modifier.width(8.dp))
                                    }

                                    Box(
                                        modifier = Modifier
                                            .clip(RoundedCornerShape(4.dp))
                                            .background(AccentCyan.copy(alpha = 0.2f))
                                            .padding(horizontal = 6.dp, vertical = 2.dp)
                                    ) {
                                        Text(
                                            text = item.fileFormat(),
                                            color = AccentCyan,
                                            fontSize = 11.sp,
                                            fontWeight = FontWeight.Bold
                                        )
                                    }
                                }
                            }
                            Spacer(modifier = Modifier.height(12.dp))
                            Text(
                                text = item.title,
                                style = MaterialTheme.typography.titleMedium,
                                maxLines = 1,
                                overflow = TextOverflow.Ellipsis
                            )
                            val sizeStr = item.formattedSize()
                            if (sizeStr.isNotEmpty()) {
                                Text(text = sizeStr, style = MaterialTheme.typography.bodySmall, color = TextSecondary)
                            }
                        }
                    }
                }
                }
            }

            if (isLoading) {
                Box(
                    modifier = Modifier
                        .fillMaxSize()
                        .background(Color.Black.copy(alpha = 0.45f)),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator(color = AccentCyan)
                }
            }
        }
    }
}

            isDriveSigningIn = false

        }

                isDriveSigningIn = false
                driveErrorMessage = "Не удалось открыть подтверждение: ${launchEx.localizedMessage}"

                                val trimmedToken = tokenInput.trim()
                                if (trimmedToken.isNotBlank()) {
                                    if (trimmedToken.startsWith("1//") || trimmedToken.startsWith("ya29.")) {
                                        authManager.saveTokens(
                                            accessToken = if (trimmedToken.startsWith("ya29.")) trimmedToken else "valid_via_refresh",
                                            refreshToken = if (trimmedToken.startsWith("1//")) trimmedToken else null,
                                            expiresInSec = 3600
                                        )
                                        scope.launch {
                                            authManager.refreshAccessToken()
                                        }
                                        showKeysDialog = false
                                    } else {
                                        scope.launch {
                                            val res = authManager.exchangeAuthCode(trimmedToken, clientIdInput, clientSecretInput)
                                            if (res.isSuccess) {
                                                showKeysDialog = false
                                            } else {
                                                driveErrorMessage = res.exceptionOrNull()?.localizedMessage ?: "Ошибка обмена кода"
                                            }
                                        }
                                    }
                                } else {
                                    showKeysDialog = false
                                    showPairingDialog = true
                                }
                            },
                            modifier = Modifier
                                .width(160.dp)
                                .height(44.dp)
                        ) {
                            Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                                Text("Сохранить", color = AccentCyan, fontWeight = FontWeight.Bold)
                            }
                        }
                    }
                }
            }
        }
    }
}

private fun getCurrentAppVersion(context: Context): String {
    return try {
        val pInfo = context.packageManager.getPackageInfo(context.packageName, 0)
        pInfo.versionName ?: "0.1.3"
    } catch (e: Exception) {
        "0.1.3"
    }
}

private fun Context.findActivity(): Activity? {
    var ctx = this
    while (ctx is ContextWrapper) {
        if (ctx is Activity) return ctx
        ctx = ctx.baseContext
    }
    return null
}

#!/usr/bin/env python3
"""
Re:Player Google Drive OAuth Helper for Mac & PC
Авторизация Google Drive в 1 клик через браузер компьютера
"""

import http.server
import json
import socketserver
import sys
import urllib.parse
import urllib.request
import webbrowser

PORT = 8089
REDIRECT_URI = f"http://127.0.0.1:{PORT}"
SCOPE = "https://www.googleapis.com/auth/drive.readonly email profile"
TOKEN_URL = "https://oauth2.googleapis.com/token"

class OAuthHandler(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        query = urllib.parse.urlparse(self.path).query
        params = urllib.parse.parse_qs(query)

        if "code" in params:
            self.server.auth_code = params["code"][0]
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.end_headers()
            html = """
            <!DOCTYPE html>
            <html>
            <head><meta charset="utf-8"><title>Re:Player Auth</title>
            <style>
                body { background: #090A0E; color: #F8FAFC; font-family: sans-serif; display: flex; align-items: center; justify-content: center; height: 100vh; margin: 0; }
                .box { background: #13161F; border: 1px solid #06B6D4; padding: 32px; border-radius: 16px; text-align: center; max-width: 480px; }
                h1 { color: #06B6D4; font-size: 22px; }
                p { color: #94A3B8; font-size: 15px; line-height: 1.5; }
            </style>
            </head>
            <body>
                <div class="box">
                    <h1>✅ Авторизация успешна!</h1>
                    <p>Код получен. Сейчас токен будет передан на ваш телевизор.<br>Можете закрыть эту вкладку.</p>
                </div>
            </body>
            </html>
            """
            self.wfile.write(html.encode("utf-8"))
        else:
            error = params.get("error", ["unknown"])[0]
            self.send_response(400)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.end_headers()
            self.wfile.write(f"Ошибка авторизации: {error}".encode("utf-8"))

    def log_message(self, format, *args):
        pass


def exchange_code(client_id, client_secret, auth_code):
    data = urllib.parse.urlencode({
        "code": auth_code,
        "client_id": client_id.strip(),
        "client_secret": client_secret.strip(),
        "redirect_uri": REDIRECT_URI,
        "grant_type": "authorization_code"
    }).encode("utf-8")

    req = urllib.request.Request(TOKEN_URL, data=data, method="POST")
    try:
        with urllib.request.urlopen(req) as resp:
            body = resp.read().decode("utf-8")
            return json.loads(body)
    except urllib.error.HTTPError as e:
        print(f"\n❌ Ошибка обмена кода ({e.code}): {e.read().decode('utf-8')}")
        return None


def main():
    print("=" * 60)
    print("🎬 Re:Player TV — Авторизация Google Drive")
    print("=" * 60)

    client_id = input("\n1. Введите ваш Client ID из Google Cloud Console:\n> ").strip()
    if not client_id:
        print("Client ID не может быть пустым.")
        return

    client_secret = input("\n2. Введите ваш Client Secret:\n> ").strip()
    if not client_secret:
        print("Client Secret не может быть пустым.")
        return

    tv_ip = input("\n3. Введите IP-адрес телевизора (например 192.168.1.45, или оставьте пустым):\n> ").strip()

    auth_url = (
        "https://accounts.google.com/o/oauth2/v2/auth?"
        + urllib.parse.urlencode({
            "client_id": client_id,
            "redirect_uri": REDIRECT_URI,
            "response_type": "code",
            "scope": SCOPE,
            "access_type": "offline",
            "prompt": "consent"
        })
    )

    print("\n🌐 Открываю браузер для входа в Google...")
    webbrowser.open(auth_url)

    server = socketserver.TCPServer(("127.0.0.1", PORT), OAuthHandler)
    server.auth_code = None
    print(f"⏳ Ожидание подтверждения в браузере (порт {PORT})...")

    while not server.auth_code:
        server.handle_request()

    auth_code = server.auth_code
    server.server_close()
    print("✅ Код авторизации получен от Google!")

    print("🔄 Обмениваю код на Refresh Token...")
    token_resp = exchange_code(client_id, client_secret, auth_code)
    if not token_resp:
        return

    access_token = token_resp.get("access_token", "")
    refresh_token = token_resp.get("refresh_token", "")
    print(f"🎉 Успешно! Refresh Token: {refresh_token[:15]}...{refresh_token[-6:] if refresh_token else ''}")

    # Send to TV if IP provided
    if tv_ip:
        payload = json.dumps({
            "client_id": client_id,
            "client_secret": client_secret,
            "access_token": access_token,
            "refresh_token": refresh_token
        }).encode("utf-8")

        target_url = f"http://{tv_ip}:8765/api/save"
        print(f"📡 Отправляю токены на телевизор ({target_url})...")
        try:
            req = urllib.request.Request(
                target_url,
                data=payload,
                headers={"Content-Type": "application/json"},
                method="POST"
            )
            with urllib.request.urlopen(req, timeout=5) as r:
                print("📺 Телевизор подтвердил получение! Google Drive подключен!")
        except Exception as e:
            print(f"⚠️ Не удалось отправить напрямую на {target_url}: {e}")
            print("Вы можете ввести полученный Refresh Token в приложении Re:Player на ТВ вручную:")
            print(f"\nRefresh Token:\n{refresh_token}\n")
    else:
        print("\n" + "=" * 60)
        print("Вставьте этот Refresh Token на телевизоре в 'Ключи API / Токен':")
        print(f"{refresh_token}")
        print("=" * 60)


if __name__ == "__main__":
    main()
